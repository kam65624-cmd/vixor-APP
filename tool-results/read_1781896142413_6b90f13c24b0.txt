   218→function DiscoverWatchlist() {
   219→  const { t } = useI18n();
   220→  const queryClient = useQueryClient();
   221→
   222→  // Selected watchlist state
   223→  const [selectedWatchlistId, setSelectedWatchlistId] = useState<string | null>(null);
   224→
   225→  // Dialog states
   226→  const [showCreateDialog, setShowCreateDialog] = useState(false);
   227→  const [newWatchlistName, setNewWatchlistName] = useState("");
   228→  const [showRenameDialog, setShowRenameDialog] = useState(false);
   229→  const [renameWatchlistName, setRenameWatchlistName] = useState("");
   230→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   231→
   232→  // Fetch all watchlists with their items
   233→  const fetchWatchlists = useStableServerFn(getWatchlists);
   234→  const fetchPrices = useStableServerFn(getMarketPrices);
   235→  const addFn = useStableServerFn(addToWatchlist);
   236→  const removeFn = useStableServerFn(removeFromWatchlist);
   237→  const updateFn = useStableServerFn(updateWatchlistItem);
   238→  const createFn = useStableServerFn(createWatchlist);
   239→  const deleteFn = useStableServerFn(deleteWatchlist);
   240→  const renameFn = useStableServerFn(renameWatchlist);
   241→
   242→  const { data: watchlists = [], isLoading: watchlistsLoading } = useQuery(
   243→    useMemo(
   244→      () => ({
   245→        queryKey: ["user-watchlists"] as const,
   246→        queryFn: () => fetchWatchlists({}),
   247→        staleTime: 30_000,
   248→      }),
   249→      [fetchWatchlists],
   250→    ),
   251→  );
   252→
   253→  const { data: prices = [], isLoading: pricesLoading } = useQuery(
   254→    useMemo(
   255→      () => ({
   256→        queryKey: ["market-prices"] as const,
   257→        queryFn: () => fetchPrices({}),
   258→        staleTime: 30_000,
   259→        refetchInterval: 60_000,
   260→      }),
   261→      [fetchPrices],
   262→    ),
   263→  );
   264→
   265→  // Build price map for quick lookup
   266→  const priceMap = useMemo(() => {
   267→    const map = new Map<string, any>();
   268→    for (const p of prices) {
   269→      map.set(p.pair, p);
   270→    }
   271→    return map;
   272→  }, [prices]);
   273→
   274→  // Derive the current watchlist from selected ID or default
   275→  const currentWatchlist = useMemo(() => {
   276→    if (!watchlists || watchlists.length === 0) return null;
   277→    if (selectedWatchlistId) {
   278→      return watchlists.find((w: any) => w.id === selectedWatchlistId) ?? null;
   279→    }
   280→    // Auto-select default watchlist
   281→    const defaultWl = watchlists.find((w: any) => w.is_default) ?? watchlists[0];
   282→    return defaultWl;
   283→  }, [watchlists, selectedWatchlistId]);
   284→
   285→  const items = (currentWatchlist as any)?.items ?? [];
   286→  const addedPairs = new Set(items.map((i: any) => i.pair));
   287→
   288→  // Create watchlist mutation
   289→  const createMutation = useMutation({
   290→    mutationFn: (vars: { name: string; isDefault?: boolean }) =>
   291→      createFn({ data: vars }),
   292→    onSuccess: (data: any) => {
   293→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   294→      setShowCreateDialog(false);
   295→      setNewWatchlistName("");
   296→      if (data?.id) {
   297→        setSelectedWatchlistId(data.id);
   298→      }
   299→    },
   300→  });
   301→
   302→  // Delete watchlist mutation
   303→  const deleteMutation = useMutation({
   304→    mutationFn: (vars: { watchlistId: string }) =>
   305→      deleteFn({ data: vars }),
   306→    onSuccess: () => {
   307→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   308→      setSelectedWatchlistId(null);
   309→      setShowDeleteDialog(false);
   310→    },
   311→  });
   312→
   313→  // Rename watchlist mutation
   314→  const renameMutation = useMutation({
   315→    mutationFn: (vars: { watchlistId: string; name: string }) =>
   316→      renameFn({ data: vars }),
   317→    onSuccess: () => {
   318→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   319→      setShowRenameDialog(false);
   320→      setRenameWatchlistName("");
   321→    },
   322→  });
   323→
   324→  // Add mutation
   325→  const addMutation = useMutation({
   326→    mutationFn: (vars: { watchlistId?: string; pair: string; category: string }) =>
   327→      addFn({ data: vars }),
   328→    onSuccess: () => {
   329→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   330→    },
   331→  });
   332→
   333→  // Remove mutation
   334→  const removeMutation = useMutation({
   335→    mutationFn: (vars: { itemId: string }) =>
   336→      removeFn({ data: vars }),
   337→    onSuccess: () => {
   338→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   339→    },
   340→  });
   341→
   342→  // Update notes mutation
   343→  const updateMutation = useMutation({
   344→    mutationFn: (vars: { itemId: string; notes?: string; sortOrder?: number }) =>
   345→      updateFn({ data: vars }),
   346→    onSuccess: () => {
   347→      queryClient.invalidateQueries({ queryKey: ["user-watchlists"] });
   348→    },
   349→  });
   350→
   351→  // Category filter for adding pairs
   352→  const [categoryFilter, setCategoryFilter] = useState<CategoryFilter>("all");
   353→  const [searchQuery, setSearchQuery] = useState("");
   354→  const [editingNotes, setEditingNotes] = useState<string | null>(null);
   355→  const [notesValue, setNotesValue] = useState("");
   356→
   357→  const filteredPairs = useMemo(() => {
   358→    let filtered = AVAILABLE_PAIRS;
   359→    if (categoryFilter !== "all") {
   360→      filtered = filtered.filter((p) => p.category === categoryFilter);
   361→    }
   362→    if (searchQuery.trim()) {
   363→      const q = searchQuery.toLowerCase();
   364→      filtered = filtered.filter((p) => p.pair.toLowerCase().includes(q));
   365→    }
   366→    return filtered;
   367→  }, [categoryFilter, searchQuery]);
   368→
   369→  const getCategoryBadge = (cat: string) => {
   370→    const styles: Record<string, string> = {
   371→      forex: "bg-primary/10 text-primary border-primary/20",
   372→      crypto: "bg-amber-500/10 text-amber-600 border-amber-500/20",
   373→      commodity: "bg-orange-500/10 text-orange-600 border-orange-500/20",
   374→      stocks: "bg-cyan-500/10 text-cyan-600 border-cyan-500/20",
   375→    };
   376→    return styles[cat] ?? styles.forex;
   377→  };
   378→
   379→  const handleAddPair = (pair: string, category: string) => {
   380→    addMutation.mutate({
   381→      pair,
   382→      category,
   383→      watchlistId: currentWatchlist?.id,
   384→    });
   385→  };
   386→
   387→  const handleRemove = (itemId: string) => {
   388→    removeMutation.mutate({ itemId });
   389→  };
   390→
   391→  const handleSaveNotes = (itemId: string) => {
   392→    updateMutation.mutate({ itemId, notes: notesValue });
   393→    setEditingNotes(null);
   394→  };
   395→
   396→  const handleMoveUp = (item: any, index: number) => {
   397→    if (index === 0) return;
   398→    const prevItem = items[index - 1];
   399→    updateMutation.mutate({ itemId: item.id, sortOrder: prevItem.sort_order });
   400→    updateMutation.mutate({ itemId: prevItem.id, sortOrder: item.sort_order });
   401→  };
   402→
   403→  const handleMoveDown = (item: any, index: number) => {
   404→    if (index === items.length - 1) return;
   405→    const nextItem = items[index + 1];
   406→    updateMutation.mutate({ itemId: item.id, sortOrder: nextItem.sort_order });
   407→    updateMutation.mutate({ itemId: nextItem.id, sortOrder: item.sort_order });
   408→  };
   409→
   410→  const handleCreateWatchlist = () => {
   411→    if (!newWatchlistName.trim()) return;
   412→    createMutation.mutate({ name: newWatchlistName.trim() });
   413→  };
   414→
   415→  const handleDeleteWatchlist = () => {
   416→    if (!currentWatchlist) return;
   417→    deleteMutation.mutate({ watchlistId: currentWatchlist.id });
   418→  };
   419→
   420→  const handleRenameWatchlist = () => {
   421→    if (!currentWatchlist || !renameWatchlistName.trim()) return;
   422→    renameMutation.mutate({ watchlistId: currentWatchlist.id, name: renameWatchlistName.trim() });
   423→  };
   424→
   425→  const isLoading = watchlistsLoading || pricesLoading;
   426→
   427→  return (
   428→    <div className="space-y-4 animate-in fade-in slide-in-from-bottom-2 duration-300">
   429→      {/* Header with Watchlist Switcher */}
   430→      <div className="flex items-center justify-between gap-2">
   431→        <div className="flex items-center gap-2 min-w-0 flex-1">
   432→          {/* Watchlist Switcher Dropdown */}
   433→          <DropdownMenu>
   434→            <DropdownMenuTrigger asChild>
   435→              <button className="flex items-center gap-1.5 min-w-0 group">
   436→                <h2 className="font-bold text-lg truncate group-hover:text-primary transition-colors">
   437→                  {currentWatchlist?.name ?? t("discover.yourWatchlist")}
   438→                </h2>
   439→                <ChevronDown className="size-4 text-muted-foreground group-hover:text-primary transition-colors shrink-0" />
   440→              </button>
   441→            </DropdownMenuTrigger>
   442→            <DropdownMenuContent align="start" className="w-56">
   443→              {watchlists.map((wl: any) => (
   444→                <DropdownMenuItem
   445→                  key={wl.id}
   446→                  onClick={() => setSelectedWatchlistId(wl.id)}
   447→                  className={currentWatchlist?.id === wl.id ? "bg-primary/10" : ""}
   448→                >
   449→                  <span className="truncate">{wl.name}</span>
   450→                  {wl.is_default && (
   451→                    <span className="ml-auto text-[9px] font-bold text-muted-foreground">
   452→                      {t("discover.default") ?? "Default"}
   453→                    </span>
   454→                  )}
   455→                </DropdownMenuItem>
   456→              ))}
   457→              <DropdownMenuSeparator />
   458→              <DropdownMenuItem onClick={() => setShowCreateDialog(true)}>
   459→                <ListPlus className="size-3.5 mr-2" />
   460→                <span>{t("discover.newWatchlist") ?? "New Watchlist"}</span>
   461→              </DropdownMenuItem>
   462→            </DropdownMenuContent>
   463→          </DropdownMenu>
   464→        </div>
   465→
   466→        {/* Watchlist Actions */}
   467→        <div className="flex items-center gap-1 shrink-0">
   468→          {currentWatchlist && (
   469→            <DropdownMenu>
   470→              <DropdownMenuTrigger asChild>
   471→                <button className="size-8 rounded-lg flex items-center justify-center bg-card border border-border hover:border-primary/30 hover:text-primary transition-colors">
   472→                  <MoreHorizontal className="size-4" />
   473→                </button>
   474→              </DropdownMenuTrigger>
   475→              <DropdownMenuContent align="end">
   476→                <DropdownMenuItem
   477→                  onClick={() => {
   478→                    setRenameWatchlistName(currentWatchlist.name ?? "");
   479→                    setShowRenameDialog(true);
   480→                  }}
   481→                >
   482→                  <Pencil className="size-3.5 mr-2" />
   483→                  <span>{t("discover.renameWatchlist") ?? "Rename"}</span>
   484→                </DropdownMenuItem>
   485→                {!currentWatchlist.is_default && (
   486→                  <>
   487→                    <DropdownMenuSeparator />
   488→                    <DropdownMenuItem
   489→                      onClick={() => setShowDeleteDialog(true)}
   490→                      className="text-bearish focus:text-bearish"
   491→                    >
   492→                      <Trash2 className="size-3.5 mr-2" />
   493→                      <span>{t("discover.deleteWatchlist") ?? "Delete Watchlist"}</span>
   494→                    </DropdownMenuItem>
   495→                  </>
   496→                )}
   497→              </DropdownMenuContent>
   498→            </DropdownMenu>
   499→          )}
   500→          <button
   501→            onClick={() => setShowCreateDialog(true)}
   502→            className="size-8 rounded-lg flex items-center justify-center bg-primary/10 text-primary border border-primary/20 hover:bg-primary/20 transition-colors"
   503→            title={t("discover.newWatchlist") ?? "New Watchlist"}
   504→          >
   505→            <Plus className="size-4" />
   506→          </button>
   507→        </div>
   508→      </div>
   509→
   510→      {/* Subtitle with item count */}
   511→      {currentWatchlist && items.length > 0 && (
   512→        <p className="text-xs text-muted-foreground -mt-2">
   513→          {items.length} {items.length === 1 ? "pair" : "pairs"}
   514→          {!currentWatchlist.is_default && (
   515→            <span className="ml-2 text-[9px] font-bold px-1.5 py-0.5 rounded border bg-card border-border text-muted-foreground">
   516→              {currentWatchlist.name}
   517→            </span>
   518→          )}
   519→        </p>
   520→      )}
   521→
   522→      {/* Watchlist Items */}
   523→      {isLoading ? (
   524→        <div className="space-y-3">
   525→          {[1, 2, 3].map((i) => (
   526→            <div key={i} className="vixor-card p-4">
   527→              <div className="flex items-center justify-between">
   528→                <div className="space-y-2 flex-1">
   529→                  <Skeleton className="h-5 w-20" />
   530→                  <Skeleton className="h-4 w-24" />
   531→                </div>
   532→                <Skeleton className="h-8 w-16 rounded" />
   533→              </div>
   534→            </div>
   535→          ))}
   536→        </div>
   537→      ) : items.length === 0 ? (
   538→        /* Empty State */
   539→        <div className="vixor-card p-8 text-center border-dashed">
   540→          <Eye className="size-10 text-muted-foreground/20 mx-auto mb-3" />
   541→          <h3 className="font-bold text-sm text-foreground mb-1">{t("discover.watchlistEmpty")}</h3>
   542→          <p className="text-xs text-muted-foreground mb-4">{t("discover.addFirstPair")}</p>
   543→        </div>
   544→      ) : (
   545→        <div className="space-y-2">
   546→          {items.map((item: any, index: number) => {
   547→            const priceData = priceMap.get(item.pair);
   548→            const change = priceData?.change24h ?? 0;
   549→            const isPositive = change >= 0;
   550→
   551→            return (
   552→              <div
   553→                key={item.id}
   554→                className="vixor-card p-3 group hover:border-primary/30 transition-colors"
   555→              >
   556→                <div className="flex items-center gap-3">

  1297→function formatTimeAgo(timestamp: number) {
  1298→  const diff = Date.now() - timestamp;
  1299→  const mins = Math.floor(diff / 60000);
  1300→  if (mins < 1) return "Just now";
  1301→  if (mins < 60) return `${mins}m ago`;
  1302→  const hours = Math.floor(mins / 60);
  1303→  if (hours < 24) return `${hours}h ago`;
  1304→  const days = Math.floor(hours / 24);
  1305→  return `${days}d ago`;
  1306→}
