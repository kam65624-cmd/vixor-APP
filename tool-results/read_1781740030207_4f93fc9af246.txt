    27→// ---------- CREATE ANALYSIS (image-based) ----------
    28→export const createAnalysis = createServerFn({ method: "POST" })
    29→  .middleware([requireSupabaseAuth])
    30→  .validator((d: unknown) => CreateAnalysisInput.parse(d))
    31→  .handler(async ({ data, context }) => {
    32→    const { userId, supabase } = context;
    33→    const { supabaseAdmin } = await import("@/shared/supabase/client.server");
    34→    const { runChartAnalysis } = await import("@/domains/analysis/server/run-analysis");
    35→
    36→    // Decode base64
    37→    const b64 = data.imageBase64.includes(",") ? data.imageBase64.split(",")[1] : data.imageBase64;
    38→    const bytes = Uint8Array.from(Buffer.from(b64, "base64"));
    39→
    40→    // Check premium / points
    41→    const { data: sub } = await supabase
    42→      .from("premium_subscriptions")
    43→      .select("id")
    44→      .eq("user_id", userId)
    45→      .gt("current_period_end", new Date().toISOString())
    46→      .limit(1)
    47→      .maybeSingle();
    48→    const isPremium = !!sub;
    49→    if (!isPremium) {
    50→      const { data: bal } = await supabase
    51→        .from("points_balances")
    52→        .select("balance")
    53→        .eq("user_id", userId)
    54→        .maybeSingle();
    55→      if (bal && bal.balance < 10) throw new Error("INSUFFICIENT_POINTS");
    56→    }
    57→
    58→    // Try to upload to storage (optional — won't block analysis if it fails)
    59→    let imagePath: string | null = null;
    60→    try {
    61→      const ext = data.mimeType.split("/")[1] === "jpeg" ? "jpg" : data.mimeType.split("/")[1];
    62→      const path = `${userId}/${crypto.randomUUID()}.${ext}`;
    63→      const { error: upErr } = await supabase.storage
    64→        .from("charts")
    65→        .upload(path, bytes, { contentType: data.mimeType });
    66→      if (!upErr) imagePath = path;
    67→      else console.warn("[Storage] Upload skipped:", upErr.message);
    68→    } catch (storageErr) {
    69→      console.warn("[Storage] Upload error (non-fatal):", storageErr);
    70→    }
    71→
    72→    // Insert queued row
    73→    const { data: row, error: insErr } = await supabase
    74→      .from("analyses")
    75→      .insert({ user_id: userId, image_path: imagePath, status: "processing" })
    76→      .select("id")
    77→      .single();
    78→    if (insErr || !row) throw new Error(insErr?.message ?? "insert failed");
    79→
    80→    // Run AI analysis
    81→    try {
    82→      let realBars: import("@/domains/analysis/engine/core/types").OHLCVBar[] | undefined;
    83→      try {
    84→        const pair = data.selectedPair || "EUR/USD";
    85→        const tf =
    86→          data.tradingStyle === "Scalping"
    87→            ? "15M"
    88→            : data.tradingStyle === "Swing Trading"
    89→              ? "4H"
    90→              : "1H";
    91→
    92→        // ── Try multiple data sources with fallback ──
    93→        const { fetchBinanceKlines, fetchTwelveDataKlines } = await import("@/domains/market/server/price-fetcher");
    94→
    95→        // Source 1: Binance for crypto pairs
    96→        if (
    97→          pair.includes("USDT") ||
    98→          pair.includes("BTC") ||
    99→          pair.includes("ETH") ||
   100→          pair.includes("SOL")
   101→        ) {
   102→          try {
   103→            const klines = await fetchBinanceKlines(pair, tf, 200);
   104→            if (klines.length > 20) {
   105→              realBars = klines.map((k) => ({
   106→                time: k.time,
   107→                open: k.open,
   108→                high: k.high,
   109→                low: k.low,
   110→                close: k.close,
   111→                volume: k.volume,
   112→              }));
   113→              console.log(`[Vixor] Using ${realBars.length} real Binance candles for ${pair}/${tf}`);
   114→            }
   115→          } catch (err) {
   116→            console.warn(`[Vixor] Binance fetch failed for ${pair}:`, err instanceof Error ? err.message : String(err));
   117→          }
   118→        }
   119→
   120→        // Source 2: TwelveData for forex/commodity pairs
   121→        if (!realBars) {
   122→          try {
   123→            const klines = await fetchTwelveDataKlines(pair, tf, 200);
   124→            if (klines.length > 20) {
   125→              realBars = klines.map((k) => ({
   126→                time: k.time,
   127→                open: k.open,
   128→                high: k.high,
   129→                low: k.low,
   130→                close: k.close,
   131→                volume: k.volume,
   132→              }));
   133→              console.log(
   134→                `[Vixor] Using ${realBars.length} real TwelveData candles for ${pair}/${tf}`,
   135→              );
   136→            }
   137→          } catch (err) {
   138→            console.warn(`[Vixor] TwelveData fetch failed for ${pair}:`, err instanceof Error ? err.message : String(err));
   139→          }
   140→        }
   141→
   142→        // Source 3: Try Binance as fallback even for non-crypto (some forex pairs exist)
   143→        if (!realBars && !pair.includes("USDT")) {
   144→          try {
   145→            const klines = await fetchBinanceKlines(pair, tf, 200);
   146→            if (klines.length > 20) {
   147→              realBars = klines.map((k) => ({
   148→                time: k.time,
   149→                open: k.open,
   150→                high: k.high,
   151→                low: k.low,
   152→                close: k.close,
   153→                volume: k.volume,
   154→              }));
   155→              console.log(`[Vixor] Using ${realBars.length} Binance fallback candles for ${pair}/${tf}`);
   156→            }
   157→          } catch {
   158→            // Non-fatal
   159→          }
   160→        }
   161→
   162→        // Source 4: Try TwelveData with 1D interval as last resort
   163→        if (!realBars) {
   164→          try {
   165→            const klines = await fetchTwelveDataKlines(pair, "1D", 100);
   166→            if (klines.length > 10) {
   167→              realBars = klines.map((k) => ({
   168→                time: k.time,
   169→                open: k.open,
   170→                high: k.high,
   171→                low: k.low,
   172→                close: k.close,
   173→                volume: k.volume,
   174→              }));
   175→              console.log(`[Vixor] Using ${realBars.length} TwelveData daily candles as fallback for ${pair}`);
   176→            }
   177→          } catch {
   178→            // Non-fatal
   179→          }
   180→        }
   181→      } catch (fetchErr) {
   182→        console.warn(
   183→          "[Vixor] Failed to fetch real OHLCV data:",
   184→          fetchErr instanceof Error ? fetchErr.message : String(fetchErr),
   185→        );
   186→      }
   187→
   188→      // SOFT CHECK: If real OHLCV data is unavailable, the engine will use its
   189→      // built-in synthetic data generator as fallback. This ensures the app always
   190→      // produces analysis results, even when market data APIs are down.
   191→      if (!realBars) {
   192→        console.warn(`[Vixor] No real OHLCV data available for ${data.selectedPair || "EUR/USD"}. Engine will use synthetic data fallback.`);
   193→      }
   194→
   195→      const result = await runChartAnalysis(
   196→        bytes,
   197→        data.mimeType,
   198→        data.fileName,
   199→        data.selectedPair,
   200→        data.tradingStyle,
   201→        realBars || undefined,
   202→      );
   203→
   204→      const updateData = {
   205→        status: "complete" as const,
   206→        pair: result.pair,
   207→        timeframe: result.timeframe,
   208→        trend: result.trend,
   209→        risk_level: result.risk_level,
   210→        risk_reasons: result.risk_reasons,
   211→        invalidation_level: result.invalidation_level,
   212→        liquidity_zones: result.liquidity_zones as any,
   213→        market_structure: result.market_structure as any,
   214→        key_levels: result.key_levels as any,
   215→        recommendation: result.recommendation,
   216→        confidence: Math.round(result.confidence),
   217→        entry: result.entry,
   218→        stop_loss: result.stop_loss,
   219→        take_profit: result.take_profit,
   220→        rr: result.rr,
   221→        pattern: result.pattern,
   222→        reasons: result.reasons,
   223→        scenarios: result.scenarios as any,
   224→        management: result.management,
   225→        news: (result as any).news_impact,
   226→        raw_ai_response: result as any,
   227→      };
   228→
   229→      try {
   230→        await supabaseAdmin
   231→          .from("analyses")
   232→          .update({
   233→            ...updateData,
   234→            signal_badge: (result as any).signal_badge as any,
   235→            vixor_message: (result as any).vixor_message as any,
   236→          })
   237→          .eq("id", row.id)
   238→          .throwOnError();
   239→      } catch (colErr: any) {
   240→        if (
   241→          String(colErr?.message || "").includes("signal_badge") ||
   242→          String(colErr?.message || "").includes("vixor_message")
   243→        ) {
   244→          console.warn(
   245→            "[Vixor] signal_badge/vixor_message columns not found, storing in raw_ai_response only",
   246→          );
   247→          await supabaseAdmin.from("analyses").update(updateData).eq("id", row.id).throwOnError();
   248→        } else {
   249→          throw colErr;
   250→        }
   251→      }
   252→
   253→      if (!isPremium) {
   254→        void supabase.rpc("spend_points", {
   255→          _user: userId,
   256→          _amount: 10,
   257→          _reason: "analysis_cost",
   258→          _meta: { analysis_id: row.id },
   259→        });
   260→      }
   261→
   262→      // Reward XP (non-fatal)
   263→      void supabase
   264→        .from("profiles")
   265→        .select("xp")
   266→        .eq("id", userId)
   267→        .maybeSingle()
   268→        .then(({ data: profile }) => {
   269→          if (profile) {
   270→            void supabase
   271→              .from("profiles")
   272→              .update({ xp: ((profile as any).xp || 0) + 10 })
   273→              .eq("id", userId);
   274→          }
   275→        });
   276→    } catch (e) {
   277→      const rawMsg = e instanceof Error ? e.message : String(e);
   278→
   279→      // ── Friendly error rewriting ──
   280→      // The local SMC/ICT engine NEVER refuses to analyze — it always produces
   281→      // a result (using synthetic data as fallback if real OHLCV is unavailable).
   282→      // So if we ever see a "Unable to identify" error here, it's coming from
   283→      // stale code and we rewrite it to a clearer, accurate message.
   284→      let msg = rawMsg;
   285→      if (
   286→        rawMsg.includes("Unable to identify the asset") ||
   287→        rawMsg.includes("insufficient accuracy")
   288→      ) {
   289→        msg =
   290→          "We couldn't extract clear chart details from your image, but the analysis engine still ran on real market data for the selected pair. Try uploading a clearer screenshot, or pick the pair manually above.";
   291→      }
   292→
   293→      void supabaseAdmin
   294→        .from("analyses")
   295→        .update({ status: "failed", error_message: msg })
   296→        .eq("id", row.id);
   297→      throw new Error(msg);
   298→    }
   299→
   300→    return { id: row.id };
   301→  });

   303→// ---------- GET ANALYSIS ----------
   304→export const getAnalysis = createServerFn({ method: "GET" })
   305→  .middleware([requireSupabaseAuth])
   306→  .validator((d: unknown) => z.object({ id: z.string().uuid() }).parse(d))
   307→  .handler(async ({ data, context }) => {
   308→    const { supabase } = context;
   309→    let a: any;
   310→    try {
   311→      const { data: fullRow, error: fullErr } = await supabase
   312→        .from("analyses")
   313→        .select(
   314→          "id,user_id,image_path,status,pair,timeframe,trend,risk_level,risk_reasons,invalidation_level,liquidity_zones,market_structure,key_levels,recommendation,confidence,entry,stop_loss,take_profit,rr,pattern,reasons,scenarios,management,news,raw_ai_response,source,signal_badge,vixor_message,created_at,updated_at,error_message",
   315→        )
   316→        .eq("id", data.id)
   317→        .maybeSingle();
   318→      if (
   319→        fullErr &&
   320→        (fullErr.message.includes("signal_badge") || fullErr.message.includes("vixor_message"))
   321→      ) {
   322→        const { data: partialRow, error: partialErr } = await supabase
   323→          .from("analyses")
   324→          .select(
   325→            "id,user_id,image_path,status,pair,timeframe,trend,risk_level,risk_reasons,invalidation_level,liquidity_zones,market_structure,key_levels,recommendation,confidence,entry,stop_loss,take_profit,rr,pattern,reasons,scenarios,management,news,raw_ai_response,source,created_at,updated_at,error_message",
   326→          )
   327→          .eq("id", data.id)
   328→          .maybeSingle();
   329→        if (partialErr) throw new Error(partialErr.message);
   330→        a = partialRow;
   331→      } else if (fullErr) {
   332→        throw new Error(fullErr.message);
   333→      } else {
   334→        a = fullRow;
   335→      }
   336→    } catch {
   337→      const { data: fallbackRow, error: fbErr } = await supabase
   338→        .from("analyses")
   339→        .select(
   340→          "id,user_id,image_path,status,pair,timeframe,trend,risk_level,risk_reasons,invalidation_level,liquidity_zones,market_structure,key_levels,recommendation,confidence,entry,stop_loss,take_profit,rr,pattern,reasons,scenarios,management,news,raw_ai_response,source,created_at,updated_at,error_message",
   341→        )
   342→        .eq("id", data.id)
   343→        .maybeSingle();
   344→      if (fbErr) throw new Error(fbErr.message);
   345→      a = fallbackRow;
   346→    }
   347→    if (!a) throw new Error("Not found");
   348→
   349→    let signalBadge = (a as any)?.signal_badge ?? null;
   350→    let vixorMessage = (a as any)?.vixor_message ?? null;
   351→
   352→    if (!signalBadge && (a as any).raw_ai_response) {
   353→      try {
   354→        const raw = (a as any).raw_ai_response;
   355→        if (typeof raw === "object" && raw !== null) {
   356→          signalBadge = raw.signal_badge ?? null;
   357→          vixorMessage = raw.vixor_message ?? null;
   358→        }
   359→      } catch {
   360→        // Non-fatal
   361→      }
   362→    }
   363→
   364→    let imageUrl: string | null = null;
   365→    if ((a as any).image_path) {
   366→      const { data: signed } = await supabase.storage
   367→        .from("charts")
   368→        .createSignedUrl((a as any).image_path, 3600);
   369→      imageUrl = signed?.signedUrl ?? null;
   370→    }
   371→    return { ...a, signal_badge: signalBadge, vixor_message: vixorMessage, imageUrl } as any;
   372→  });

   374→// ---------- LIST ANALYSES (with pagination) ----------
   375→export const listAnalyses = createServerFn({ method: "GET" })
   376→  .middleware([requireSupabaseAuth])
   377→  .validator((d: unknown) =>
   378→    z
   379→      .object({
   380→        limit: z.number().min(1).max(100).default(20),

   383→      .parse(d ?? {}),
   384→  )
   385→  .handler(async ({ data, context }) => {

   390→      })
   391→      .order("created_at", { ascending: false })
   392→      .range(data.offset, data.offset + data.limit - 1);
   393→    const total = count ?? 0;
   394→    return {
   395→      items: rows ?? [],
   396→      total,
   397→      hasMore: data.offset + data.limit < total,
   398→    };
   399→  });
