    36→type ExperimentStatus = "running" | "completed" | "failed" | "cancelled";
    37→
    38→/** Shape returned from Supabase `experiments` table via listExperiments. */
    39→interface ExperimentRecord {
    40→  id: string;
    41→  user_id: string;
    42→  config: {
    43→    name: string;
    44→    assetSymbol: string;
    45→    timeframe: string;
    46→    strategyTemplate: string;
    47→    generations: number;
    48→    populationSize: number;
    49→    createdAt: string;
    50→  };
    51→  result: Record<string, unknown> | null;
    52→  status: ExperimentStatus;
    53→  created_at: string;
    54→  completed_at: string | null;
    55→}
    56→
    57→/** Derived best-score summary from the experiment result. */
    58→interface BestScoreSummary {
    59→  overall: number;
    60→  grade: string;
    61→  totalReturn: number;
    62→  maxDrawdown: number;
    63→  sharpe: number;
    64→}
    65→
    66→/** Extract a BestScoreSummary from the serialized experiment result. */
    67→function extractBestScore(result: Record<string, unknown> | null): BestScoreSummary | null {
    68→  if (!result) return null;
    69→  const best = result.bestStrategy as Record<string, unknown> | undefined;
    70→  if (!best || !best.score) return null;
    71→  const summary = best.summary as Record<string, unknown> | undefined;
    72→  const overall = typeof best.score === "number" ? best.score : 0;
    73→  const totalReturn = (summary?.totalReturn as number) ?? 0;
    74→  const maxDrawdown = (summary?.maxDrawdown as number) ?? 0;
    75→  const sharpe = (summary?.sharpe as number) ?? 0;
    76→  const grade = overall >= 90 ? "A" : overall >= 80 ? "B" : overall >= 70 ? "C" : overall >= 60 ? "D" : "F";
    77→  return { overall, grade, totalReturn, maxDrawdown, sharpe };
    78→}
    79→
    80→/** Extract elapsed time in ms from result. */
    81→function extractElapsed(result: Record<string, unknown> | null): number | null {
    82→  if (!result) return null;
    83→  const ms = result.elapsedMs as number | undefined;
    84→  return typeof ms === "number" ? ms : null;
    85→}
    86→
    87→/** Extract ranked strategies count from result. */
    88→function extractRankedCount(result: Record<string, unknown> | null): number {
    89→  if (!result) return 0;
    90→  const arr = result.rankedStrategies as unknown[] | undefined;
    91→  return Array.isArray(arr) ? arr.length : 0;
    92→}

   579→                    onClick={() => setNewTimeframe(tf)}

   581→                      newTimeframe === tf
   582→                        ? "bg-primary text-primary-foreground border-primary glow-primary"

   584→                    }`}
   585→                  >
   586→                    {tf}
   587→                  </button>
   588→                ))}
   589→              </div>
   590→            </div>
   591→          </div>
   592→
   593→          {/* Strategy */}
   594→          <div>

   601→                  key={s.id}
   602→                  onClick={() => setNewStrategy(s.id)}

   604→                    newStrategy === s.id
   605→                      ? "bg-primary text-primary-foreground border-primary glow-primary"

   636→                min={4}

   638→                value={newPopulation}

   642→            </div>

   644→

   647→            disabled={creating || !newName.trim() || !hasEnoughPoints}

   652→                <Loader2 className="size-4 animate-spin" />

   663→        </div>

   665→
   666→      {/* Create error */}

   669→          <div className="text-xs text-bearish">{createError}</div>

   672→
   673→      {/* Summary stats */}
   674→      <div className="vixor-card p-4">

   679→            </div>

   684→          </div>
   685→          <div className="p-2 rounded-lg bg-background">
   686→            <div className="text-[9px] uppercase tracking-wider text-muted-foreground font-bold">
   687→              {useT("experiments.completed") || "Completed"}
   688→            </div>
   689→            <div className="text-lg font-bold font-mono text-bullish">{completedCount}</div>
   690→          </div>
   691→          <div className="p-2 rounded-lg bg-background">
   692→            <div className="text-[9px] uppercase tracking-wider text-muted-foreground font-bold">
   693→              {useT("experiments.bestScore") || "Best Score"}
   694→            </div>
   695→            <div className="text-lg font-bold font-mono text-primary">
   696→              {bestOverall ? (
   697→                <>
   698→                  {bestOverall.overall}
   699→                  <span className="text-xs text-muted-foreground ml-1">{bestOverall.grade}</span>
   700→                </>
   701→              ) : (
   702→                "---"
   703→              )}
   704→            </div>
   705→          </div>
   706→        </div>
   707→      </div>
   708→
   709→      {/* Experiment list */}
   710→      <div className="space-y-3">
   711→        {experiments.length === 0 ? (
   712→          <div className="vixor-card p-6 text-center">
   713→            <FlaskConical className="size-8 text-muted-foreground/30 mx-auto mb-2" />
   714→            <div className="text-sm text-muted-foreground mb-2">{useT("experiments.noExperiments") || "No experiments yet. Create one to get started."}</div>
   715→            <button
   716→              onClick={() => setShowNewForm(true)}
   717→              className="px-4 h-9 rounded-xl gradient-primary text-primary-foreground text-xs font-bold glow-primary"
   718→            >
   719→              <Plus className="size-3.5 inline mr-1" />
   720→              {useT("experiments.createFirst") || "Create Your First Experiment"}
   721→            </button>
   722→          </div>
   723→        ) : (
   724→          experiments.map((exp) => <ExperimentCard key={exp.id} experiment={exp} />)
   725→        )}
   726→      </div>
   727→    </div>
   728→  );
