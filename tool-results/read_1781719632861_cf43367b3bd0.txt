    53→// SMC/ICT term highlighter
    54→function highlightSMC(text: string): React.ReactNode[] {
    55→  const smcTerms = [
    56→    "Order Block",
    57→    "Fair Value Gap",
    58→    "FVG",
    59→    "Liquidity",
    60→    "BOS",
    61→    "ChoCh",
    62→    "CHOCH",
    63→    "ICT",
    64→    "SMC",
    65→    "Sweep",
    66→    "Mitigation",
    67→    "Break of Structure",
    68→    "Change of Character",
    69→    "Imbalance",
    70→    "Premium",
    71→    "Discount",
    72→    "OB",
    73→    "NWOG",
    74→    "NDOG",
    75→  ];
    76→  const regex = new RegExp(
    77→    `(${smcTerms.map((t) => t.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")).join("|")})`,
    78→    "gi",
    79→  );
    80→  const parts = text.split(regex);
    81→  return parts.map((part, i) => {
    82→    if (smcTerms.some((t) => t.toLowerCase() === part.toLowerCase())) {
    83→      return (
    84→        <span key={i} className="text-primary font-bold bg-primary/10 px-0.5 rounded">
    85→          {part}
    86→        </span>
    87→      );
    88→    }
    89→    return part;
    90→  });
    91→}

   942→function relTime(iso: string): string {
   943→  const diff = (Date.now() - new Date(iso).getTime()) / 1000;
   944→  if (diff < 60) return "Just now";
   945→  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
   946→  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
   947→  return `${Math.floor(diff / 86400)}d ago`;
   948→}
