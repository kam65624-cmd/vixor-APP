    35→type ExperimentStatus = "running" | "completed" | "failed" | "cancelled";
    36→
    37→/** Shape returned from Supabase `experiments` table via listExperiments. */
    38→interface ExperimentRecord {
    39→  id: string;
    40→  user_id: string;
    41→  config: {
    42→    name: string;
    43→    assetSymbol: string;
    44→    timeframe: string;
    45→    strategyTemplate: string;
    46→    generations: number;
    47→    populationSize: number;
    48→    createdAt: string;
    49→  };
    50→  result: Record<string, unknown> | null;
    51→  status: ExperimentStatus;
    52→  created_at: string;
    53→  completed_at: string | null;
    54→}
    55→
    56→/** Derived best-score summary from the experiment result. */
    57→interface BestScoreSummary {
    58→  overall: number;
    59→  grade: string;
    60→  totalReturn: number;
    61→  maxDrawdown: number;
    62→  sharpe: number;
    63→}
    64→
    65→/** Extract a BestScoreSummary from the serialized experiment result. */
    66→function extractBestScore(result: Record<string, unknown> | null): BestScoreSummary | null {
    67→  if (!result) return null;
    68→  const best = result.bestStrategy as Record<string, unknown> | undefined;
    69→  if (!best || !best.score) return null;
    70→  const summary = best.summary as Record<string, unknown> | undefined;
    71→  const overall = typeof best.score === "number" ? best.score : 0;
    72→  const totalReturn = (summary?.totalReturn as number) ?? 0;
    73→  const maxDrawdown = (summary?.maxDrawdown as number) ?? 0;
    74→  const sharpe = (summary?.sharpe as number) ?? 0;
    75→  const grade = overall >= 90 ? "A" : overall >= 80 ? "B" : overall >= 70 ? "C" : overall >= 60 ? "D" : "F";
    76→  return { overall, grade, totalReturn, maxDrawdown, sharpe };
    77→}
    78→
    79→/** Extract elapsed time in ms from result. */
    80→function extractElapsed(result: Record<string, unknown> | null): number | null {
    81→  if (!result) return null;
    82→  const ms = result.elapsedMs as number | undefined;
    83→  return typeof ms === "number" ? ms : null;
    84→}
    85→
    86→/** Extract ranked strategies count from result. */
    87→function extractRankedCount(result: Record<string, unknown> | null): number {
    88→  if (!result) return 0;
    89→  const arr = result.rankedStrategies as unknown[] | undefined;
    90→  return Array.isArray(arr) ? arr.length : 0;
    91→}
