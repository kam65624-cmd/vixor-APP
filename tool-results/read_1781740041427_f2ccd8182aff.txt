    61→/**
    62→ * Run the full Vixor local analysis pipeline.
    63→ *
    64→ * This function is the drop-in replacement for `runChartAnalysis` from
    65→ * `run-analysis.server.ts`. It returns the same `LocalAnalysisResult` shape
    66→ * (compatible with `AnalysisSchema`).
    67→ */
    68→export function runLocalAnalysis(input: AnalysisInput): LocalAnalysisResult {
    69→  const pair = input.pair || detectPairFromImage(input.imageBytes) || "EUR/USD";
    70→  const timeframe = input.timeframe || "1H";
    71→  const config = PAIR_CONFIGS[pair] || PAIR_CONFIGS["EUR/USD"]!;
    72→  const decimals = config.decimals;
    73→
    74→  // ── Step 1: Get OHLCV data ──────────────────────────────────────────
    75→  // Flow:
    76→  //   1. Real bars provided (>= 20) → use them (highest accuracy)
    77→  //   2. No real bars → generate synthetic bars, run full analysis, but
    78→  //      reduce confidence and tag result so user knows data is simulated
    79→  //   3. Synthetic generation fails → minimal fallback (last resort)
    80→  let usingSyntheticData = false;
    81→  let bars: OHLCVBar[];
    82→
    83→  if (input.bars && input.bars.length >= 20) {
    84→    bars = input.bars;
    85→  } else {
    86→    // No real data — try synthetic fallback
    87→    try {
    88→      bars = generateSyntheticBars(pair, timeframe, config);
    89→      usingSyntheticData = true;
    90→    } catch {
    91→      // Synthetic generation itself failed — absolute last resort
    92→      return generateFallbackResult(pair, timeframe, config);
    93→    }
    94→  }
    95→
    96→  // ── Step 2: Market structure (SMC/ICT) ──────────────────────────────
    97→  const structureResult = analyzeMarketStructure(bars);
    98→  const swingHighs = structureResult.swingPoints.filter((s) => s.type === "HIGH");
    99→  const swingLows = structureResult.swingPoints.filter((s) => s.type === "LOW");
   100→
   101→  // ── Step 3: SMC concepts ────────────────────────────────────────────
   102→  const orderBlocks = detectOrderBlocks(bars);
   103→  const fvgs = detectFVGs(bars);
   104→  const liquidityZones = detectLiquidityZones(structureResult.swingPoints, bars);
   105→  const srLevels = detectSRLevels(bars);
   106→
   107→  // ── Step 4: BOS / CHoCH details ────────────────────────────────────
   108→  const bosResult = detectRecentBOS(bars, swingHighs, swingLows);
   109→  const trendResult = determineTrendFromSwings(swingHighs, swingLows);
   110→
   111→  // Use market structure direction as primary, fall back to swing analysis
   112→  const trend: TrendDirection =
   113→    structureResult.direction !== "SIDEWAYS" ? structureResult.direction : trendResult.direction;
   114→
   115→  // ── Step 5: Pattern detection (74 candlestick + 20 chart formations + 8 harmonic) ──
   116→  const candlePatterns = detectCandlestickPatterns(bars);
   117→  const chartFormations = detectChartFormations(bars, structureResult.swingPoints);
   118→  const harmonicPatterns = detectHarmonicPatterns(bars, structureResult.swingPoints);
   119→
   120→  // ── Step 5b: Technical indicators (RSI, MACD, BB, EMA, ADX, etc.) ──
   121→  const indicators = getLatestIndicators(bars);
   122→  const rsiStatus = getRSIStatus(indicators.rsi);
   123→  const adxStrength = getADXStrength(indicators.adx);
   124→  const emaAlignment = checkEMAAlignment(
   125→    indicators.ema9,
   126→    indicators.ema21,
   127→    indicators.ema50,
   128→    indicators.ema200,
   129→  );
   130→
   131→  // ── Step 6: ATR & Support/Resistance for risk-reward ───────────────
   132→  const atrValue = atr(bars, 14);
   133→  const currentPrice = bars[bars.length - 1]!.close;
   134→  const supportLevels = srLevels
   135→    .filter((l) => l.type === "SUPPORT")
   136→    .map((l) => l.price)
   137→    .sort((a, b) => a - b);
   138→  const resistanceLevels = srLevels
   139→    .filter((l) => l.type === "RESISTANCE")
   140→    .map((l) => l.price)
   141→    .sort((a, b) => a - b);
   142→
   143→  // ── Step 7: Risk-Reward calculation ────────────────────────────────
   144→  const rr = calculateRiskReward(
   145→    currentPrice,
   146→    trend,
   147→    atrValue,
   148→    decimals,
   149→    supportLevels,
   150→    resistanceLevels,
   151→    pair,
   152→  );
   153→
   154→  // Adjust recommendation based on pattern confluence + indicators
   155→  // CRITICAL: Confluence scoring ensures consistency — the same data must produce
   156→  // the same recommendation every time. No Math.random() anywhere.
   157→  const confluence = calculateConfluenceScore(
   158→    trend,
   159→    rr.recommendation,
   160→    candlePatterns,
   161→    chartFormations,
   162→    harmonicPatterns,
   163→    rsiStatus,
   164→    adxStrength,
   165→    emaAlignment,
   166→    indicators,
   167→    currentPrice,
   168→    orderBlocks,
   169→    fvgs,
   170→  );
   171→
   172→  // Require minimum confluence to issue BUY/SELL — otherwise WAIT
   173→  // This prevents the engine from flipping between BUY and SELL on minor data changes
   174→  const MIN_CONFLUENCE_FOR_TRADE = 3; // Need at least 3 aligned signals
   175→  let finalRec: RecommendationType;
   176→
   177→  if (rr.recommendation === "WAIT" || confluence.score < MIN_CONFLUENCE_FOR_TRADE) {
   178→    finalRec = "WAIT";
   179→  } else if (confluence.direction === "BUY" && rr.recommendation === "BUY") {
   180→    finalRec = "BUY";
   181→  } else if (confluence.direction === "SELL" && rr.recommendation === "SELL") {
   182→    finalRec = "SELL";
   183→  } else {
   184→    // Confluence direction disagrees with RR recommendation → WAIT
   185→    finalRec = "WAIT";
   186→  }
   187→
   188→  // ── Step 8: Compose the result ─────────────────────────────────────
   189→  const isBullish = finalRec === "BUY";
   190→  const isBearish = finalRec === "SELL";
   191→  const isWait = finalRec === "WAIT";
   192→
   193→  // Confidence scoring — based on confluence score (deterministic, no Math.random())
   194→  // Higher confluence = higher confidence. WAIT always gets lower confidence.
   195→  // When using synthetic data, reduce confidence by 20% and cap at 70%.
   196→  let confidence =
   197→    finalRec === "WAIT" ? (trend === "NEUTRAL" ? 42 : 48) : Math.min(35 + confluence.score * 7, 92); // 3 signals=56%, 4=63%, 5=70%, 6=77%, 7=84%, 8+=92%
   198→
   199→  if (usingSyntheticData) {
   200→    confidence = Math.min(Math.max(Math.round(confidence * 0.8), 20), 70);
   201→  }
   202→
   203→  // Top liquidity zones
   204→  const buySideLiquidity = liquidityZones
   205→    .filter((z) => z.type === "BUY_SIDE")
   206→    .sort((a, b) => b.strength - a.strength)
   207→    .slice(0, 3)
   208→    .map((z) => formatPrice(z.price, decimals));
   209→
   210→  const sellSideLiquidity = liquidityZones
   211→    .filter((z) => z.type === "SELL_SIDE")
   212→    .sort((a, b) => b.strength - a.strength)
   213→    .slice(0, 3)
   214→    .map((z) => formatPrice(z.price, decimals));
   215→
   216→  // Key levels
   217→  const keyResistance = resistanceLevels.slice(0, 3).map((p) => formatPrice(p, decimals));
   218→  const keySupport = supportLevels.slice(0, 3).map((p) => formatPrice(p, decimals));
   219→  const pivot = srLevels.find((l) => l.type === "PIVOT");
   220→  const pivotPrice = pivot ? formatPrice(pivot.price, decimals) : undefined;
   221→
   222→  // Pattern summary (includes harmonic patterns)
   223→  const primaryPattern = buildPatternSummary(
   224→    candlePatterns,
   225→    chartFormations,
   226→    harmonicPatterns,
   227→    trend,
   228→  );
   229→
   230→  // Reasons — now include confluence signals for meaningful, specific analysis
   231→  const reasons = buildReasons(
   232→    trend,
   233→    finalRec,
   234→    structureResult,
   235→    orderBlocks,
   236→    fvgs,
   237→    candlePatterns,
   238→    chartFormations,
   239→    harmonicPatterns,
   240→    srLevels,
   241→    rsiStatus,
   242→    adxStrength,
   243→    confluence,
   244→  );
   245→
   246→  // Tag synthetic data in reasons so the user is informed
   247→  if (usingSyntheticData) {
   248→    reasons.unshift("Analysis based on simulated data — results are approximate.");
   249→  }
   250→
   251→  // Scenarios
   252→  const entry = formatPrice(rr.entry, decimals);
   253→  const sl = formatPrice(rr.stopLoss, decimals);
   254→  const risk = Math.abs(rr.entry - rr.stopLoss);
   255→
   256→  const scenarios = {
   257→    conservative: {
   258→      name: "Retest Entry",
   259→      probability: 55,
   260→      entry: String(entry),
   261→      sl: formatPrice(rr.stopLoss, decimals),
   262→      tp1: formatPrice(rr.takeProfits[0] ?? rr.entry + risk, decimals),
   263→      tp2: formatPrice(rr.takeProfits[1] ?? rr.entry + risk * 1.5, decimals),
   264→      rr: "1:1.0",
   265→    },
   266→    balanced: {
   267→      name: "Market Entry",
   268→      probability: 30,
   269→      entry: String(entry),
   270→      sl: formatPrice(rr.stopLoss, decimals),
   271→      tp1: formatPrice(rr.takeProfits[1] ?? rr.entry + risk * 2, decimals),
   272→      tp2: formatPrice(rr.takeProfits[2] ?? rr.entry + risk * 2.5, decimals),
   273→      rr: rr.rr,
   274→    },
   275→    aggressive: {
   276→      name: "Breakout",
   277→      probability: 15,
   278→      entry: String(formatPrice(rr.takeProfits[0] ?? currentPrice, decimals)),
   279→      sl: formatPrice(rr.entry, decimals),
   280→      tp1: formatPrice(rr.takeProfits[2] ?? rr.entry + risk * 3, decimals),
   281→      tp2: formatPrice((rr.takeProfits[2] ?? rr.entry + risk * 3) + risk * 0.5, decimals),
   282→      rr: "1:3.0",
   283→    },
   284→  };
   285→
   286→  // Risk assessment
   287→  const riskLevel = assessRisk(trend, atrValue, currentPrice, srLevels, fvgs, orderBlocks);
   288→  const riskReasons = buildRiskReasons(riskLevel, trend, atrValue, currentPrice, srLevels);
   289→
   290→  // Management instructions
   291→  const management = buildManagementInstructions(
   292→    finalRec,
   293→    rr.entry,
   294→    rr.stopLoss,
   295→    rr.takeProfits,
   296→    rr.invalidationLevel,
   297→    decimals,
   298→  );
   299→
   300→  // Vixor message
   301→  let vixorMessage = generateVixorMessage(
   302→    pair,
   303→    timeframe,
   304→    trend,
   305→    finalRec,
   306→    confidence,
   307→    primaryPattern,
   308→    structureResult,
   309→    orderBlocks,
   310→    fvgs,
   311→  );
   312→
   313→  if (usingSyntheticData) {
   314→    vixorMessage = `[SIMULATED DATA] ${vixorMessage} Note: This analysis is based on simulated price data, not live market data. Treat with appropriate caution.`;
   315→  }
   316→
   317→  // News impact — only include if real news data is available
   318→  // Previously used deterministic fake news — now returns undefined to avoid fabricated headlines
   319→  const newsImpact: LocalAnalysisResult["news_impact"] | undefined = undefined;
   320→
   321→  // Signal badge
   322→  const signalBadge = {
   323→    direction: finalRec,
   324→    entry: String(entry),
   325→    stop_loss: String(sl),
   326→    take_profit: String(formatPrice(rr.takeProfits[1] ?? currentPrice, decimals)),
   327→    rr: rr.rr,
   328→  };
   329→
   330→  return {
   331→    pair,
   332→    timeframe,
   333→    trend: trend as TrendDirection,
   334→    risk_level: riskLevel,
   335→    risk_reasons: riskReasons,
   336→    invalidation_level: formatPrice(rr.invalidationLevel, decimals),
   337→    liquidity_zones: {
   338→      buySide: buySideLiquidity,
   339→      sellSide: sellSideLiquidity,
   340→    },
   341→    market_structure: {
   342→      direction: (structureResult.direction === "SIDEWAYS"
   343→        ? "NEUTRAL"
   344→        : structureResult.direction) as TrendDirection,
   345→      structure: structureResult.structure,
   346→      bos: structureResult.lastBOS?.price
   347→        ? formatPrice(structureResult.lastBOS.price, decimals)
   348→        : undefined,
   349→    },
   350→    key_levels: {
   351→      resistance: keyResistance,
   352→      support: keySupport,
   353→      pivot: pivotPrice,
   354→    },
   355→    recommendation: finalRec,
   356→    confidence,
   357→    entry: formatPrice(rr.entry, decimals),
   358→    stop_loss: formatPrice(rr.stopLoss, decimals),
   359→    take_profit: rr.takeProfits.map((p) => formatPrice(p, decimals)),
   360→    rr: rr.rr,
   361→    pattern: primaryPattern,
   362→    reasons,
   363→    scenarios,
   364→    management,
   365→    news_impact: newsImpact,
   366→    signal_badge: signalBadge,
   367→    vixor_message: vixorMessage,
   368→  };
   369→}

   380→// ---------------------------------------------------------------------------
   381→// Helper: Confluence scoring — requires multiple independent signals to agree
   382→// ---------------------------------------------------------------------------
   383→
   384→interface ConfluenceResult {
   385→  direction: "BUY" | "SELL" | "NONE";
   386→  score: number; // Number of independent signals that agree
   387→  signals: string[]; // Names of agreeing signals for transparency
   388→}
   389→
   390→/**
   391→ * Calculate confluence score from multiple independent analysis signals.
   392→ * This is the KEY to consistency: we don't just look at one indicator,
   393→ * we require at least 3 independent signals to agree before issuing a trade.
   394→ *
   395→ * Signals counted:
   396→ * 1. Market structure trend (HH+HL = bullish, LH+LL = bearish)
   397→ * 2. Risk-Reward direction (BUY or SELL from ATR-based levels)
   398→ * 3. Candlestick patterns (recent bullish/bearish patterns)
   399→ * 4. Chart formations (triangles, channels, etc.)
   400→ * 5. Harmonic patterns (Gartley, Bat, etc.)
   401→ * 6. RSI status (overbought → bearish, oversold → bullish)
   402→ * 7. ADX trend strength (strong trend = confirms direction)
   403→ * 8. EMA alignment (9/21/50/200)
   404→ * 9. MACD histogram direction
   405→ * 10. Order block confluence (unmitigated OBs in direction)
   406→ * 11. FVG confluence (unfilled gaps in direction)
   407→ * 12. Price position relative to key EMAs
   408→ */
   409→function calculateConfluenceScore(
   410→  trend: TrendDirection,
   411→  rrDirection: RecommendationType,
   412→  patterns: Array<{ type: string; reliability: number }>,
   413→  formations: Array<{ type: string; reliability: number }>,
   414→  harmonics: Array<{ type: string; reliability: number }>,
   415→  rsiStatus: string,
   416→  adxStrength: string,
   417→  emaAlignment: { alignment: string; strength: number },
   418→  indicators: {
   419→    rsi: number;
   420→    macdHistogram: number;
   421→    ema9: number;
   422→    ema21: number;
   423→    ema50: number;
   424→    ema200: number;
   425→    adx: number;
   426→    bollingerPosition: number;
   427→    stochK: number;
   428→    stochD: number;
   429→    volumeTrend: string;
   430→    emaTrend: string;
   431→  },
   432→  currentPrice: number,
   433→  obs: Array<{ type: string; mitigated: boolean; strength: number }>,
   434→  fvgs: Array<{ type: string; filled: boolean }>,
   435→): ConfluenceResult {
   436→  let bullSignals = 0;
   437→  let bearSignals = 0;
   438→  const bullReasons: string[] = [];
   439→  const bearReasons: string[] = [];
   440→
   441→  // 1. Market structure trend
   442→  if (trend === "BULLISH") {
   443→    bullSignals++;
   444→    bullReasons.push("Bullish market structure (HH+HL)");
   445→  }
   446→  if (trend === "BEARISH") {
   447→    bearSignals++;
   448→    bearReasons.push("Bearish market structure (LH+LL)");
   449→  }
   450→
   451→  // 2. Risk-Reward direction
   452→  if (rrDirection === "BUY") {
   453→    bullSignals++;
   454→    bullReasons.push("RR calculation favors long");
   455→  }
   456→  if (rrDirection === "SELL") {
   457→    bearSignals++;
   458→    bearReasons.push("RR calculation favors short");
   459→  }
   460→
   461→  // 3. Candlestick patterns (last 5)
   462→  const recentPatterns = patterns.slice(0, 5);
   463→  let patternBull = 0,
   464→    patternBear = 0;
   465→  for (const p of recentPatterns) {
   466→    if (p.type === "BULLISH") patternBull += p.reliability / 100;
   467→    if (p.type === "BEARISH") patternBear += p.reliability / 100;
   468→  }
   469→  if (patternBull > patternBear + 0.3) {
   470→    bullSignals++;
   471→    bullReasons.push("Bullish candlestick patterns");
   472→  }
   473→  if (patternBear > patternBull + 0.3) {
   474→    bearSignals++;
   475→    bearReasons.push("Bearish candlestick patterns");
   476→  }
   477→
   478→  // 4. Chart formations
   479→  const recentFormations = formations.slice(0, 3);
   480→  let formBull = 0,
   481→    formBear = 0;
   482→  for (const f of recentFormations) {
   483→    if (f.type === "BULLISH") formBull += f.reliability / 100;
   484→    if (f.type === "BEARISH") formBear += f.reliability / 100;
   485→  }
   486→  if (formBull > formBear) {
   487→    bullSignals++;
   488→    bullReasons.push("Bullish chart formation");
   489→  }
   490→  if (formBear > formBull) {
   491→    bearSignals++;
   492→    bearReasons.push("Bearish chart formation");
   493→  }
   494→
   495→  // 5. Harmonic patterns (high weight)
   496→  const recentHarmonics = harmonics.slice(0, 2);
   497→  let harmBull = 0,
   498→    harmBear = 0;
   499→  for (const h of recentHarmonics) {
   500→    if (h.type === "BULLISH") harmBull += (h.reliability / 100) * 1.5;
   501→    if (h.type === "BEARISH") harmBear += (h.reliability / 100) * 1.5;
   502→  }
   503→  if (harmBull > harmBear) {
   504→    bullSignals++;
   505→    bullReasons.push("Bullish harmonic pattern");
   506→  }
   507→  if (harmBear > harmBull) {
   508→    bearSignals++;
   509→    bearReasons.push("Bearish harmonic pattern");
   510→  }
   511→
   512→  // 6. RSI status
   513→  if (rsiStatus === "OVERSOLD") {
   514→    bullSignals++;
   515→    bullReasons.push("RSI oversold — reversal zone");
   516→  }
   517→  if (rsiStatus === "OVERBOUGHT") {
   518→    bearSignals++;
   519→    bearReasons.push("RSI overbought — reversal zone");
   520→  }
   521→
   522→  // 7. ADX strength — only counts if there's a direction
   523→  if (adxStrength === "STRONG" && trend === "BULLISH") {
   524→    bullSignals++;
   525→    bullReasons.push("Strong ADX confirms bullish trend");
   526→  }
   527→  if (adxStrength === "STRONG" && trend === "BEARISH") {
   528→    bearSignals++;
   529→    bearReasons.push("Strong ADX confirms bearish trend");
   530→  }
   531→  if (adxStrength === "WEAK") {
   532→    // Weak ADX reduces confidence in any direction — no signal added
   533→  }
   534→
   535→  // 8. EMA alignment
   536→  if (emaAlignment.alignment === "BULLISH" && emaAlignment.strength >= 60) {
   537→    bullSignals++;
   538→    bullReasons.push(`Bullish EMA alignment (${emaAlignment.strength}%)`);
   539→  }
   540→  if (emaAlignment.alignment === "BEARISH" && emaAlignment.strength >= 60) {
   541→    bearSignals++;
   542→    bearReasons.push(`Bearish EMA alignment (${emaAlignment.strength}%)`);
   543→  }
   544→
   545→  // 9. MACD histogram direction (deterministic based on value)
   546→  const macdH = indicators.macdHistogram;
   547→  if (!isNaN(macdH)) {
   548→    if (macdH > 0) {
   549→      bullSignals++;
   550→      bullReasons.push("MACD histogram positive");
   551→    }
   552→    if (macdH < 0) {
   553→      bearSignals++;
   554→      bearReasons.push("MACD histogram negative");
   555→    }
   556→  }
   557→
   558→  // 10. Order block confluence
   559→  const unmitigated = obs.filter((o) => !o.mitigated);
   560→  const bullOBs = unmitigated.filter((o) => o.type === "BULLISH").length;
   561→  const bearOBs = unmitigated.filter((o) => o.type === "BEARISH").length;
   562→  if (bullOBs > bearOBs + 1) {
   563→    bullSignals++;
   564→    bullReasons.push(`${bullOBs} unmitigated bullish OBs`);
   565→  }
   566→  if (bearOBs > bullOBs + 1) {
   567→    bearSignals++;
   568→    bearReasons.push(`${bearOBs} unmitigated bearish OBs`);
   569→  }
   570→
   571→  // 11. FVG confluence
   572→  const activeFVGs = fvgs.filter((f) => !f.filled);
   573→  const bullFVGs = activeFVGs.filter((f) => f.type === "BULLISH").length;
   574→  const bearFVGs = activeFVGs.filter((f) => f.type === "BEARISH").length;
   575→  if (bullFVGs > bearFVGs) {
   576→    bullSignals++;
   577→    bullReasons.push(`${bullFVGs} unfilled bullish FVGs`);
   578→  }
   579→  if (bearFVGs > bullFVGs) {
   580→    bearSignals++;
   581→    bearReasons.push(`${bearFVGs} unfilled bearish FVGs`);
   582→  }
   583→
   584→  // 12. Price position relative to EMA200 (if available)
   585→  const ema200 = indicators.ema200;
   586→  if (!isNaN(ema200)) {
   587→    if (currentPrice > ema200) {
   588→      bullSignals++;
   589→      bullReasons.push("Price above EMA200");
   590→    }
   591→    if (currentPrice < ema200) {
   592→      bearSignals++;
   593→      bearReasons.push("Price below EMA200");
   594→    }
   595→  }
   596→
   597→  // Determine dominant direction
   598→  if (bullSignals > bearSignals) {
   599→    return { direction: "BUY", score: bullSignals, signals: bullReasons };
   600→  } else if (bearSignals > bullSignals) {
   601→    return { direction: "SELL", score: bearSignals, signals: bearReasons };
   602→  } else {
   603→    return { direction: "NONE", score: 0, signals: [] };
   604→  }
   605→}

   611→function calculateConfidence(
   612→  trend: TrendDirection,
   613→  rec: RecommendationType,
   614→  patterns: Array<{ type: string; reliability: number }>,
   615→  formations: Array<{ type: string; reliability: number }>,
   616→  harmonics: Array<{ type: string; reliability: number }>,
   617→  obs: Array<{ type: string; mitigated: boolean; strength: number }>,
   618→  fvgs: Array<{ type: string; filled: boolean }>,
   619→  structure: { direction: string; bosEvents: Array<{ type: string }> },
   620→  rsiStatus: string,
   621→  adxStrength: string,
   622→  emaAlignment: { alignment: string; strength: number },
   623→): number {
   624→  if (rec === "WAIT") {
   625→    // Deterministic confidence for WAIT based on trend clarity
   626→    // No Math.random() — must be reproducible across runs
   627→    const waitConfidence = trend === "NEUTRAL" ? 42 : 48;
   628→    return waitConfidence;
   629→  }
   630→
   631→  let confidence = 55;
   632→
   633→  // Trend alignment
   634→  if ((trend === "BULLISH" && rec === "BUY") || (trend === "BEARISH" && rec === "SELL")) {
   635→    confidence += 10;
   636→  }
   637→
   638→  // Recent patterns support the direction
   639→  const supportingPatterns = patterns.filter(
   640→    (p) => (rec === "BUY" && p.type === "BULLISH") || (rec === "SELL" && p.type === "BEARISH"),
   641→  );
   642→  confidence += Math.min(supportingPatterns.length * 3, 10);
   643→
   644→  // Chart formations
   645→  const supportingFormations = formations.filter(
   646→    (f) => (rec === "BUY" && f.type === "BULLISH") || (rec === "SELL" && f.type === "BEARISH"),
   647→  );
   648→  confidence += Math.min(supportingFormations.length * 5, 10);
   649→
   650→  // Harmonic patterns (carry more weight)
   651→  const supportingHarmonics = harmonics.filter(
   652→    (h) => (rec === "BUY" && h.type === "BULLISH") || (rec === "SELL" && h.type === "BEARISH"),
   653→  );
   654→  confidence += Math.min(supportingHarmonics.length * 7, 12);
   655→
   656→  // Unmitigated order blocks in the direction
   657→  const unmitigatedOBs = obs.filter(
   658→    (ob) =>
   659→      !ob.mitigated &&
   660→      ((rec === "BUY" && ob.type === "BULLISH") || (rec === "SELL" && ob.type === "BEARISH")),
   661→  );
   662→  confidence += Math.min(unmitigatedOBs.length * 3, 8);
   663→
   664→  // Active FVGs
   665→  const activeFVGs = fvgs.filter((f) => !f.filled);
   666→  confidence += Math.min(activeFVGs.length, 4);
   667→
   668→  // BOS/CHoCH events
   669→  if (structure.bosEvents.length > 0) confidence += 3;
   670→  if (structure.bosEvents.some((e) => e.type === "CHoCH")) confidence += 2;
   671→
   672→  // Indicator-based confidence adjustments
   673→  // RSI in favorable zone
   674→  if (
   675→    (rec === "BUY" && rsiStatus === "OVERSOLD") ||
   676→    (rec === "SELL" && rsiStatus === "OVERBOUGHT")
   677→  ) {
   678→    confidence += 5; // Reversal from extreme RSI
   679→  } else if (
   680→    (rec === "BUY" && rsiStatus === "OVERBOUGHT") ||
   681→    (rec === "SELL" && rsiStatus === "OVERSOLD")
   682→  ) {
   683→    confidence -= 5; // Trading against RSI extreme
   684→  }
   685→
   686→  // ADX trend strength
   687→  if (adxStrength === "STRONG") confidence += 5;
   688→  else if (adxStrength === "WEAK") confidence -= 3;
   689→
   690→  // EMA alignment
   691→  if (emaAlignment.alignment === (rec === "BUY" ? "BULLISH" : "BEARISH")) {
   692→    confidence += Math.min(emaAlignment.strength / 10, 5);
   693→  } else if (emaAlignment.alignment !== "NEUTRAL") {
   694→    confidence -= 3;
   695→  }
   696→
   697→  return Math.min(Math.max(confidence, 40), 95);
   698→}

   737→function buildReasons(
   738→  trend: TrendDirection,
   739→  rec: RecommendationType,
   740→  structure: { direction: string; structure: string; lastBOS?: { type: string } },
   741→  obs: Array<{ type: string; mitigated: boolean }>,
   742→  fvgs: Array<{ type: string; filled: boolean }>,

   747→  rsiStatus: string,
   748→  adxStrength: string,
   749→  confluence: ConfluenceResult,

   751→  const reasons: string[] = [];
   752→
   753→  // If WAIT, explain why

   756→      reasons.push(
   757→        `Insufficient confluence — only ${confluence.score} signal(s) aligned. Need at least 3 for a trade.`,
   758→      );
   759→    } else {
   760→      reasons.push(
   761→        "Conflicting signals — bullish and bearish confluences are balanced. Wait for clear direction.",
   762→      );
   763→    }
   764→    if (trend === "NEUTRAL") {
   765→      reasons.push(
   766→        "Market is consolidating with no clear trend structure. Wait for a BOS or CHoCH.",
   767→      );
   768→    }
   769→    if (adxStrength === "WEAK") {
   770→      reasons.push("ADX shows weak trend momentum — no directional conviction from institutions.");
   771→    }
   772→    return reasons.slice(0, 5);
   773→  }
   774→
   775→  // Use confluence signals as the primary reasons — these are the REAL reasons
   776→  // the engine is making this call, not generic SMC terminology
   777→  const relevantSignals = rec === "BUY" ? confluence.signals : confluence.signals;
   778→  for (let i = 0; i < Math.min(relevantSignals.length, 3); i++) {
   779→    reasons.push(relevantSignals[i]!);
   780→  }
   781→
   782→  // Add SMC-specific context if BOS/CHoCH exists
   783→  if (structure.lastBOS?.type === "CHoCH") {
   784→    reasons.push("Change of Character (CHoCH) confirms institutional reversal intent.");
   785→  } else if (structure.lastBOS?.type === "BOS" && reasons.length < 5) {
   786→    reasons.push("Break of Structure (BOS) confirms trend continuation.");
   787→  }
   788→
   789→  // Ensure at least 3 reasons
   790→  if (reasons.length < 3) {
   791→    if (trend === "BULLISH") {
   792→      reasons.push("Price is showing higher-timeframe bullish structure with orderly pullbacks.");
   793→    } else if (trend === "BEARISH") {
   794→      reasons.push("Price is showing lower-timeframe bearish structure with orderly pullbacks.");
   795→    }
   796→  }
   797→
   798→  return reasons.slice(0, 5);
   799→}

   804→
   805→function assessRisk(
   806→  trend: TrendDirection,
   807→  atrValue: number,
   808→  currentPrice: number,
   809→  srLevels: Array<{ type: string; price: number; strength: number }>,
   810→  fvgs: Array<{ filled: boolean }>,
   811→  obs: Array<{ mitigated: boolean }>,
   812→): RiskLevel {
   813→  let riskScore = 0;
   814→
   815→  // Sideways = high risk
   816→  if (trend === "NEUTRAL") riskScore += 3;
   817→
   818→  // High volatility = higher risk
   819→  const atrPercent = (atrValue / currentPrice) * 100;
   820→  if (atrPercent > 2) riskScore += 2;
   821→  if (atrPercent > 3) riskScore += 1;
   822→
   823→  // No clear S/R levels = higher risk
   824→  const nearLevels = srLevels.filter((l) => Math.abs(l.price - currentPrice) / currentPrice < 0.01);
   825→  if (nearLevels.length === 0) riskScore += 1;
   826→
   827→  // Many unfilled FVGs = potential reversal risk
   828→  const activeFVGs = fvgs.filter((f) => !f.filled).length;
   829→  if (activeFVGs > 5) riskScore += 1;
   830→
   831→  // Many mitigated OBs = less institutional conviction
   832→  const mitigatedOBs = obs.filter((o) => o.mitigated).length;
   833→  if (mitigatedOBs > obs.length * 0.6) riskScore += 1;
   834→
   835→  if (riskScore >= 5) return "HIGH";
   836→  if (riskScore >= 3) return "MEDIUM";
   837→  return "LOW";
   838→}

  1119→// ---------------------------------------------------------------------------
  1120→// Helper: Generate synthetic OHLCV bars for fallback analysis
  1121→// ---------------------------------------------------------------------------
  1122→//
  1123→// This function creates realistic-looking OHLCV price data based on the pair
  1124→// config. It is used as the PRIMARY fallback when no real market data is
  1125→// available, so the engine can still produce a real (though approximate)
  1126→// analysis instead of a minimal "WAIT" result.
  1127→//
  1128→// Key properties:
  1129→//   - DETERMINISTIC: Same pair + timeframe always produces the same data
  1130→//     (seeded PRNG based on pair name). No Math.random() anywhere.
  1131→//   - REALISTIC: Random walk with volatility from pair config, evolving trend
  1132→//     bias, occasional wick sweeps, and proper OHLCV relationships.
  1133→//   - STRUCTURE-RICH: 200 bars with enough swing points, trends, and ranges
  1134→//     for the SMC/ICT engine to detect meaningful patterns.
  1135→// ---------------------------------------------------------------------------
  1136→
  1137→/**
  1138→ * Simple deterministic hash for seeding the PRNG from a string.
  1139→ * Same as the hashCode in candle-utils but kept local to avoid coupling.
  1140→ */
  1141→function hashSeed(str: string): number {
  1142→  let hash = 0;
  1143→  for (let i = 0; i < str.length; i++) {
  1144→    const char = str.charCodeAt(i);
  1145→    hash = (hash << 5) - hash + char;
  1146→    hash = hash & hash; // Convert to 32-bit integer
  1147→  }
  1148→  return Math.abs(hash);
  1149→}
  1150→
  1151→/**
  1152→ * Generate 200 synthetic OHLCV bars from the pair config.
  1153→ *
  1154→ * Uses a Lehmer / LCG PRNG seeded by `pair` so the same pair always
  1155→ * generates the same bar sequence — essential for deterministic analysis.
  1156→ */
  1157→function generateSyntheticBars(pair: string, timeframe: string, config: PairConfig): OHLCVBar[] {
  1158→  const BAR_COUNT = 200;
  1159→
  1160→  // Seeded PRNG (Lehmer / LCG) — same algorithm as generateOHLCV in candle-utils
  1161→  let s = hashSeed(pair + timeframe);
  1162→  function next(): number {
  1163→    s = ((s * 1664525 + 1013904223) >>> 0) & 0xffffffff;
  1164→    return (s >>> 0) / 0xffffffff;
  1165→  }
  1166→
  1167→  // Timeframe multipliers for volatility scaling
  1168→  const tfMultiplier: Record<string, number> = {
  1169→    "1M": 0.15,
  1170→    "5M": 0.25,
  1171→    "15M": 0.4,
  1172→    "30M": 0.55,
  1173→    "1H": 0.7,
  1174→    "4H": 0.85,
  1175→    "1D": 1.0,
  1176→    "1W": 1.5,
  1177→  };
  1178→  const tf = tfMultiplier[timeframe] ?? 0.7;
  1179→
  1180→  // Interval in seconds for timestamp generation
  1181→  const tfMs: Record<string, number> = {
  1182→    "1M": 60_000,
  1183→    "5M": 300_000,
  1184→    "15M": 900_000,
  1185→    "30M": 1_800_000,
  1186→    "1H": 3_600_000,
  1187→    "4H": 14_400_000,
  1188→    "1D": 86_400_000,
  1189→    "1W": 604_800_000,
  1190→  };
  1191→  const intervalSec = Math.floor((tfMs[timeframe] ?? 3_600_000) / 1000);
  1192→  const startTimeSec = Math.floor(Date.now() / 1000) - BAR_COUNT * intervalSec;
  1193→
  1194→  const bars: OHLCVBar[] = [];
  1195→  let price = config.basePrice;
  1196→
  1197→  // Evolving trend bias for realistic trending / ranging behaviour
  1198→  let trendBias = 0;
  1199→  let trendDuration = 0;
  1200→  let maxTrendDuration = Math.floor(20 + next() * 40);
  1201→
  1202→  // Track recent swing levels for more realistic structure
  1203→  let lastSwingHigh = price;
  1204→  let lastSwingLow = price;
  1205→
  1206→  for (let i = 0; i < BAR_COUNT; i++) {
  1207→    // Periodically update trend bias for realistic trending / ranging behaviour
  1208→    trendDuration++;
  1209→    if (trendDuration > maxTrendDuration) {
  1210→      trendBias = (next() - 0.5) * 0.6; // range [-0.3, 0.3]
  1211→      trendDuration = 0;
  1212→      maxTrendDuration = Math.floor(20 + next() * 40);
  1213→    }
  1214→
  1215→    const vol = config.volatility * tf;
  1216→    const baseChange = (next() - 0.48 + trendBias * 0.1) * vol * price;
  1217→    const open = price;
  1218→    const close = open + baseChange;
  1219→
  1220→    // Generate wicks — occasionally create longer wicks for liquidity sweeps
  1221→    const sweepChance = next();
  1222→    const wickMultiplierUp = sweepChance > 0.92 ? 1.5 + next() * 2 : 0.5 + next() * 0.5;
  1223→    const wickMultiplierDown =
  1224→      sweepChance > 0.88 && sweepChance <= 0.92 ? 1.5 + next() * 2 : 0.5 + next() * 0.5;
  1225→
  1226→    const wickUp = next() * vol * price * wickMultiplierUp;
  1227→    const wickDown = next() * vol * price * wickMultiplierDown;
  1228→
  1229→    // Enforce proper OHLCV relationships:
  1230→    //   high >= max(open, close), low <= min(open, close)
  1231→    const high = Math.max(open, close) + wickUp;
  1232→    const low = Math.min(open, close) - wickDown;
  1233→
  1234→    // Volume — higher on big moves, occasional spikes
  1235→    const isSpike = next() > 0.95;
  1236→    const volumeBase = isSpike ? 3000 + next() * 5000 : 500 + next() * 2000;
  1237→    const volume = volumeBase * (1 + Math.abs(baseChange) / (vol * price + 1e-10));
  1238→
  1239→    bars.push({
  1240→      time: startTimeSec + i * intervalSec,
  1241→      open: Number(open.toFixed(config.decimals + 1)),
  1242→      high: Number(high.toFixed(config.decimals + 1)),
  1243→      low: Number(low.toFixed(config.decimals + 1)),
  1244→      close: Number(close.toFixed(config.decimals + 1)),
  1245→      volume: Math.floor(volume),
  1246→    });
  1247→
  1248→    // Update swing tracking
  1249→    if (close > lastSwingHigh) lastSwingHigh = close;
  1250→    if (close < lastSwingLow) lastSwingLow = close;
  1251→
  1252→    // Mean-reversion pull: soft pull toward base price to avoid unbounded drift
  1253→    const driftFromBase = (price - config.basePrice) / config.basePrice;
  1254→    price = close * (1 - driftFromBase * 0.005); // gentle mean reversion
  1255→
  1256→    // Update last swing levels slowly
  1257→    lastSwingHigh = lastSwingHigh * 0.998 + price * 0.002;
  1258→    lastSwingLow = lastSwingLow * 0.998 + price * 0.002;
  1259→  }
  1260→
  1261→  return bars;
  1262→}
