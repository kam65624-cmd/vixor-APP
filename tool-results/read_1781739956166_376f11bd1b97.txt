   936→// ═══════════════════════════════════════════════
   937→
   938→function formatTimeAgo(
   939→  timestamp: number,
   940→  t?: (key: string, params?: Record<string, string | number>) => string,
   941→) {
   942→  const diff = Date.now() - timestamp;
   943→  const mins = Math.floor(diff / 60000);
   944→  if (mins < 1) return t ? t("common.justNow") : "Just now";
   945→  if (mins < 60) return t ? t("common.mAgo", { mins }) : `${mins}m ago`;
   946→  const hours = Math.floor(mins / 60);
   947→  if (hours < 24) return t ? t("common.hAgo", { hours }) : `${hours}h ago`;
   948→  const days = Math.floor(hours / 24);
   949→  return t ? t("common.dAgo", { days }) : `${days}d ago`;
   950→}
   951→
