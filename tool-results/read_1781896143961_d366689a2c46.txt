   168→function CopilotPage() {
   169→  const { t } = useI18n();
   170→  const queryClient = useQueryClient();
   171→  const search = useSearch({ strict: false }) as {
   172→    chartPair?: string;
   173→    chartTimeframe?: string;
   174→    chartPrice?: number;
   175→    chartSymbol?: string;
   176→  };
   177→
   178→  // Build chart session context if available (from charts page navigation)
   179→  const chartSession = useMemo(() => {
   180→    if (search.chartPair && search.chartTimeframe && search.chartPrice) {
   181→      return {
   182→        pair: search.chartPair,
   183→        timeframe: search.chartTimeframe,
   184→        currentPrice: search.chartPrice,
   185→        tradingViewSymbol: search.chartSymbol || search.chartPair,
   186→      };
   187→    }
   188→    return undefined;
   189→  }, [search.chartPair, search.chartTimeframe, search.chartPrice, search.chartSymbol]);
   190→
   191→  // ─── Chat State ───
   192→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   193→  const [input, setInput] = useState("");
   194→  const [activeAgent, setActiveAgent] = useState<AgentId>("auto");
   195→  const [showAgents, setShowAgents] = useState(false);
   196→  const [consensusMode, setConsensusMode] = useState(false);
   197→  const [activeConversationId, setActiveConversationId] = useState<string | null>(null);
   198→  const messagesEndRef = useRef<HTMLDivElement>(null);
   199→  const inputRef = useRef<HTMLTextAreaElement>(null);
   200→
   201→  // ─── Sidebar State ───
   202→  const [sidebarOpen, setSidebarOpen] = useState(true);
   203→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null);
   204→  const [editingTitleId, setEditingTitleId] = useState<string | null>(null);
   205→  const [editingTitleValue, setEditingTitleValue] = useState("");
   206→  const [mobileDrawerOpen, setMobileDrawerOpen] = useState(false);
   207→
   208→  // ─── Stable Server Functions ───
   209→  const askCopilotFn = useStableServerFn(askCopilot);
   210→  const getConsensusFn = useStableServerFn(getConsensus);
   211→  const createConversationFn = useStableServerFn(createConversation);
   212→  const listConversationsFn = useStableServerFn(listConversations);
   213→  const getConversationFn = useStableServerFn(getConversation);
   214→  const saveMessageFn = useStableServerFn(saveMessage);
   215→  const deleteConversationFn = useStableServerFn(deleteConversation);
   216→  const updateConversationTitleFn = useStableServerFn(updateConversationTitle);
   217→
   218→  // ─── Fetch Conversation List (paginated) ───
   219→  const [convPage, setConvPage] = useState(1);
   220→  const CONV_PAGE_SIZE = 15;
   221→  const conversationsQuery = useQuery({
   222→    queryKey: ["copilot-conversations", convPage],
   223→    queryFn: () =>
   224→      listConversationsFn({
   225→        data: { limit: CONV_PAGE_SIZE, offset: (convPage - 1) * CONV_PAGE_SIZE },
   226→      }),
   227→  });
   228→
   229→  // ─── Copilot Mutation ───
   230→  const copilotMutation = useMutation({
   231→    mutationFn: (data: {
   232→      message: string;
   233→      history: { role: "user" | "assistant"; content: string }[];
   234→      agent: AgentId;
   235→      chartSession?: { pair: string; timeframe: string; currentPrice: number; tradingViewSymbol: string };
   236→    }) => askCopilotFn({ data }),
   237→    onSuccess: async (result) => {
   238→      const assistantMsg: ChatMessage = {
   239→        id: crypto.randomUUID(),
   240→        role: "assistant",
   241→        content: result.response,
   242→        agent: result.agent,
   243→        timestamp: Date.now(),
   244→      };
   245→      setMessages((prev) => [...prev, assistantMsg]);
   246→      setConsensusMode(false);
   247→
   248→      // Save assistant message to DB
   249→      if (activeConversationId) {
   250→        try {
   251→          await saveMessageFn({
   252→            data: {
   253→              conversation_id: activeConversationId,
   254→              role: "assistant",
   255→              content: result.response,
   256→              agent_id: result.agent,
   257→            },
   258→          });
   259→          // Refresh conversation list to pick up title changes
   260→          queryClient.invalidateQueries({ queryKey: ["copilot-conversations"] });
   261→        } catch {
   262→          // Silent fail — persistence shouldn't break the UX
   263→        }
   264→      }
   265→    },
   266→    onError: (error) => {
   267→      const errorMsg: ChatMessage = {
   268→        id: crypto.randomUUID(),
   269→        role: "assistant",
   270→        content: `⚠️ ${error.message || "Failed to get AI response. Please try again."}`,
   271→        agent: "error",
   272→        timestamp: Date.now(),
   273→      };
   274→      setMessages((prev) => [...prev, errorMsg]);
   275→      setConsensusMode(false);
   276→    },
   277→  });
   278→
   279→  // ─── Consensus Mutation ───
   280→  const consensusMutation = useMutation({
   281→    mutationFn: (data: { message: string }) => getConsensusFn({ data }),
   282→    onSuccess: async (result) => {
   283→      const consensusMsg: ChatMessage = {
   284→        id: crypto.randomUUID(),
   285→        role: "consensus",
   286→        content: "",
   287→        agent: "consensus",
   288→        consensusData: result,
   289→        timestamp: Date.now(),
   290→      };
   291→      setMessages((prev) => [...prev, consensusMsg]);
   292→      setConsensusMode(false);
   293→
   294→      // Save consensus as assistant message with metadata
   295→      if (activeConversationId) {
   296→        try {
   297→          await saveMessageFn({
   298→            data: {
   299→              conversation_id: activeConversationId,
   300→              role: "assistant",
   301→              content: result.synthesis,
   302→              agent_id: "consensus",
   303→              metadata: {
   304→                consensusData: {
   305→                  responses: result.responses,
   306→                  synthesis: result.synthesis,
   307→                },
   308→              },
   309→            },
   310→          });
   311→          queryClient.invalidateQueries({ queryKey: ["copilot-conversations"] });
   312→        } catch {
   313→          // Silent fail
   314→        }
   315→      }
   316→    },
   317→    onError: (error) => {
   318→      const errorMsg: ChatMessage = {
   319→        id: crypto.randomUUID(),
   320→        role: "assistant",
   321→        content: `⚠️ ${error.message || "Failed to get consensus. Please try again."}`,
   322→        agent: "error",
   323→        timestamp: Date.now(),
   324→      };
   325→      setMessages((prev) => [...prev, errorMsg]);
   326→      setConsensusMode(false);
   327→    },
   328→  });
   329→
   330→  // ─── Send Message ───
   331→  const sendMessage = useCallback(
   332→    async (text: string, agentOverride?: AgentId) => {
   333→      const trimmed = text.trim();
   334→      if (!trimmed || copilotMutation.isPending || consensusMutation.isPending) return;
   335→      const agent = agentOverride || activeAgent;
   336→      const userMsg: ChatMessage = {
   337→        id: crypto.randomUUID(),
   338→        role: "user",
   339→        content: trimmed,
   340→        timestamp: Date.now(),
   341→      };
   342→      setMessages((prev) => [...prev, userMsg]);
   343→      const history = messages.slice(-10).map((m) => ({ role: m.role as "user" | "assistant", content: m.content }));
   344→
   345→      // ─── Persistence: Create conversation if needed ───
   346→      let convId = activeConversationId;
   347→      if (!convId) {
   348→        try {
   349→          const conv = await createConversationFn({
   350→            data: {
   351→              agent_id: agent,
   352→              is_consensus: consensusMode,
   353→            },
   354→          });
   355→          convId = conv.id;
   356→          setActiveConversationId(convId);
   357→          // Save the user message to DB
   358→          await saveMessageFn({
   359→            data: {
   360→              conversation_id: convId,
   361→              role: "user",
   362→              content: trimmed,
   363→            },
   364→          });
   365→          // Refresh conversation list
   366→          queryClient.invalidateQueries({ queryKey: ["copilot-conversations"] });
   367→        } catch {
   368→          // If persistence fails, continue with ephemeral chat
   369→        }
   370→      } else {
   371→        // Save user message to existing conversation
   372→        try {
   373→          await saveMessageFn({
   374→            data: {
   375→              conversation_id: convId,
   376→              role: "user",
   377→              content: trimmed,
   378→            },
   379→          });
   380→        } catch {
   381→          // Silent fail
   382→        }
   383→      }
   384→
   385→      if (consensusMode) {
   386→        consensusMutation.mutate({ message: trimmed });
   387→      } else {
   388→        copilotMutation.mutate({ message: trimmed, history, agent, chartSession });
   389→      }
   390→    },
   391→    [activeAgent, copilotMutation, consensusMutation, messages, consensusMode, activeConversationId, createConversationFn, saveMessageFn, queryClient, chartSession],
   392→  );
   393→
   394→  const handleSubmit = useCallback(
   395→    (e: React.FormEvent) => {
   396→      e.preventDefault();
   397→      sendMessage(input);
   398→      setInput("");
   399→    },
   400→    [input, sendMessage],
   401→  );
   402→
   403→  const handleKeyDown = useCallback(
   404→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   405→      if (e.key === "Enter" && !e.shiftKey) {
   406→        e.preventDefault();
   407→        sendMessage(input);
   408→        setInput("");
   409→      }
   410→    },
   411→    [input, sendMessage],
   412→  );
   413→
   414→  // ─── New Chat ───
   415→  const startNewChat = useCallback(() => {
   416→    setMessages([]);
   417→    setActiveConversationId(null);
   418→    setConsensusMode(false);
   419→    setMobileDrawerOpen(false);
   420→  }, []);
   421→
   422→  // ─── Load Conversation ───
   423→  const loadConversation = useCallback(
   424→    async (convId: string) => {
   425→      try {
   426→        const conv = await getConversationFn({ data: { conversationId: convId } });
   427→        setActiveConversationId(convId);
   428→
   429→        // Convert DB messages to ChatMessage format
   430→        const loadedMessages: ChatMessage[] = (conv.messages || []).map((msg: any) => {
   431→          // Check if this is a consensus message
   432→          if (msg.agent_id === "consensus" && msg.metadata?.consensusData) {
   433→            return {
   434→              id: msg.id || crypto.randomUUID(),
   435→              dbId: msg.id,
   436→              role: "consensus",
   437→              content: "",
   438→              agent: "consensus",
   439→              consensusData: msg.metadata.consensusData,
   440→              timestamp: new Date(msg.created_at).getTime(),
   441→            };
   442→          }
   443→          return {
   444→            id: msg.id || crypto.randomUUID(),
   445→            dbId: msg.id,
   446→            role: msg.role as "user" | "assistant" | "consensus",
   447→            content: msg.content,
   448→            agent: msg.agent_id || undefined,
   449→            timestamp: new Date(msg.created_at).getTime(),
   450→          };
   451→        });
   452→        setMessages(loadedMessages);
   453→        setActiveAgent((conv.agent_id as AgentId) || "auto");
   454→        setMobileDrawerOpen(false);
   455→      } catch {
   456→        // If loading fails, just clear the chat
   457→        startNewChat();
   458→      }
   459→    },
   460→    [getConversationFn, startNewChat],
   461→  );
   462→
   463→  // ─── Delete Conversation ───
   464→  const handleDeleteConversation = useCallback(
   465→    async (convId: string) => {
   466→      try {
   467→        await deleteConversationFn({ data: { conversationId: convId } });
   468→        if (activeConversationId === convId) {
   469→          startNewChat();
   470→        }
   471→        queryClient.invalidateQueries({ queryKey: ["copilot-conversations"] });
   472→        setDeleteConfirmId(null);
   473→      } catch {
   474→        // Silent fail
   475→      }
   476→    },
   477→    [deleteConversationFn, activeConversationId, startNewChat, queryClient],
   478→  );
   479→
   480→  // ─── Rename Conversation ───
   481→  const handleRenameConversation = useCallback(
   482→    async (convId: string, newTitle: string) => {
   483→      try {
   484→        await updateConversationTitleFn({ data: { conversationId: convId, title: newTitle } });
   485→        queryClient.invalidateQueries({ queryKey: ["copilot-conversations"] });
   486→        setEditingTitleId(null);
   487→      } catch {
   488→        // Silent fail
   489→      }
   490→    },
   491→    [updateConversationTitleFn, queryClient],
   492→  );
   493→
   494→  // ─── Auto-scroll ───
   495→  useEffect(() => {
   496→    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
   497→  }, [messages]);
   498→
   499→  useEffect(() => {
   500→    inputRef.current?.focus();
   501→  }, []);
   502→
   503→  const currentAgentConfig = AGENTS.find((a) => a.id === activeAgent)!;
   504→  const isPending = copilotMutation.isPending || consensusMutation.isPending;
   505→  const conversationsRaw = conversationsQuery.data as
   506→    | { items: ConversationSummary[]; total: number; hasMore: boolean }
   507→    | undefined;
   508→  const conversations = conversationsRaw?.items ?? [];
   509→  const conversationsTotal = conversationsRaw?.total ?? 0;
   510→
   511→  // ─── Get agent icon for sidebar ───
   512→  const getAgentIcon = (agentId: string | null) => {
   513→    const agent = AGENTS.find((a) => a.id === agentId);
   514→    if (!agent || agentId === "auto") return Zap;
   515→    return agent.icon;
   516→  };
   517→
   518→  const getAgentColor = (agentId: string | null) => {
   519→    const agent = AGENTS.find((a) => a.id === agentId);
   520→    if (!agent || agentId === "auto") return "text-primary";
   521→    return agent.color;
   522→  };
   523→
   524→  return (
   525→    <div className="flex h-[calc(100vh-8rem)] animate-in fade-in slide-in-from-bottom-2 duration-500">
   526→      {/* ─── Desktop Sidebar ─── */}
   527→      <div
   528→        className={`hidden lg:flex flex-col border-r border-border transition-all duration-300 shrink-0 ${
   529→          sidebarOpen ? "w-72" : "w-0 overflow-hidden"
   530→        }`}
   531→      >
   532→        <ConversationSidebar
   533→          conversations={conversations}
   534→          activeConversationId={activeConversationId}
   535→          isLoading={conversationsQuery.isLoading}
   536→          onSelect={loadConversation}
   537→          onDelete={handleDeleteConversation}
   538→          onRename={handleRenameConversation}
   539→          onNewChat={startNewChat}
   540→          deleteConfirmId={deleteConfirmId}
   541→          setDeleteConfirmId={setDeleteConfirmId}
   542→          editingTitleId={editingTitleId}
   543→          setEditingTitleId={setEditingTitleId}
   544→          editingTitleValue={editingTitleValue}
   545→          setEditingTitleValue={setEditingTitleValue}
   546→          getAgentIcon={getAgentIcon}
   547→          getAgentColor={getAgentColor}
   548→          page={convPage}
   549→          pageSize={CONV_PAGE_SIZE}
   550→          total={conversationsTotal}
   551→          onPageChange={setConvPage}
   552→        />
   553→      </div>
   554→
   555→      {/* ─── Mobile Drawer Overlay ─── */}
   556→      {mobileDrawerOpen && (
   557→        <div className="lg:hidden fixed inset-0 z-50 flex">
   558→          <div className="absolute inset-0 bg-black/50" onClick={() => setMobileDrawerOpen(false)} />
   559→          <div className="relative w-80 max-w-[85vw] h-full bg-background border-r border-border animate-in slide-in-from-left duration-200">
   560→            <ConversationSidebar
   561→              conversations={conversations}
   562→              activeConversationId={activeConversationId}
   563→              isLoading={conversationsQuery.isLoading}
   564→              onSelect={loadConversation}
   565→              onDelete={handleDeleteConversation}
   566→              onRename={handleRenameConversation}
   567→              onNewChat={startNewChat}
   568→              deleteConfirmId={deleteConfirmId}
   569→              setDeleteConfirmId={setDeleteConfirmId}
   570→              editingTitleId={editingTitleId}
   571→              setEditingTitleId={setEditingTitleId}
   572→              editingTitleValue={editingTitleValue}
   573→              setEditingTitleValue={setEditingTitleValue}
   574→              getAgentIcon={getAgentIcon}
   575→              getAgentColor={getAgentColor}
   576→              page={convPage}
   577→              pageSize={CONV_PAGE_SIZE}
   578→              total={conversationsTotal}
   579→              onPageChange={setConvPage}
   580→            />
   581→          </div>
   582→        </div>
   583→      )}
   584→
   585→      {/* ─── Main Chat Area ─── */}
   586→      <div className="flex flex-col flex-1 min-w-0">
   587→        {/* ─── Header ─── */}
   588→        <div className="flex-shrink-0 pb-3">
   589→          <div className="flex items-center justify-between">
   590→            <div className="flex items-center gap-2">
   591→              {/* Sidebar toggle */}
   592→              <button
   593→                onClick={() => {

  1047→// ─── Relative time formatter ───
  1048→function formatRelativeTime(dateStr: string): string {
  1049→  const now = Date.now();
  1050→  const then = new Date(dateStr).getTime();
  1051→  const diff = now - then;
  1052→
  1053→  const minutes = Math.floor(diff / 60000);
  1054→  if (minutes < 1) return "Just now";
  1055→  if (minutes < 60) return `${minutes}m ago`;
  1056→
  1057→  const hours = Math.floor(minutes / 60);
  1058→  if (hours < 24) return `${hours}h ago`;
  1059→
  1060→  const days = Math.floor(hours / 24);
  1061→  if (days < 7) return `${days}d ago`;
  1062→
  1063→  return new Date(dateStr).toLocaleDateString([], {
  1064→    month: "short",
  1065→    day: "numeric",
  1066→  });
  1067→}

  1343→function FormattedContent({ content }: { content: string }) {
  1344→  const formatted = useMemo(() => {
  1345→    const blocks = content.split(/\n\n+/);
  1346→    return blocks.map((block, blockIdx) => {
  1347→      const lines = block.split("\n");
  1348→      return (
  1349→        <div key={blockIdx} className={blockIdx > 0 ? "mt-3" : ""}>
  1350→          {lines.map((line, lineIdx) => {
  1351→            if (line.startsWith("## "))
  1352→              return (
  1353→                <h3 key={lineIdx} className="text-sm font-bold text-foreground mt-2 mb-1">
  1354→                  {formatInline(line.slice(3))}
  1355→                </h3>
  1356→              );
  1357→            if (line.startsWith("# "))
  1358→              return (
  1359→                <h3 key={lineIdx} className="text-sm font-bold text-primary mt-2 mb-1">
  1360→                  {formatInline(line.slice(2))}
  1361→                </h3>
  1362→              );
  1363→            if (line.startsWith("> "))
  1364→              return (
  1365→                <div
  1366→                  key={lineIdx}
  1367→                  className="pl-3 border-l-2 border-primary/40 my-1 text-muted-foreground"
  1368→                >
  1369→                  {formatInline(line.slice(2))}
  1370→                </div>
  1371→              );
  1372→            if (line.match(/^[-*•]\s/))
  1373→              return (
  1374→                <div key={lineIdx} className="flex items-start gap-1.5 my-0.5">
  1375→                  <span className="text-primary mt-0.5 shrink-0">•</span>
  1376→                  <span>{formatInline(line.replace(/^[-*•]\s/, ""))}</span>
  1377→                </div>
  1378→              );
  1379→            if (line.match(/^\d+\.\s/)) {
  1380→              const match = line.match(/^(\d+\.)\s(.*)$/);
  1381→              if (match)
  1382→                return (
  1383→                  <div key={lineIdx} className="flex items-start gap-1.5 my-0.5">
  1384→                    <span className="text-primary font-bold text-xs mt-0.5 shrink-0">
  1385→                      {match[1]}
  1386→                    </span>
  1387→                    <span>{formatInline(match[2])}</span>
  1388→                  </div>
  1389→                );
  1390→            }
  1391→            if (line.trim())
  1392→              return (
  1393→                <div key={lineIdx} className="my-0.5">
  1394→                  {formatInline(line)}
  1395→                </div>
  1396→              );
  1397→            return null;
  1398→          })}
  1399→        </div>
  1400→      );
  1401→    });
  1402→  }, [content]);
  1403→
  1404→  return <>{formatted}</>;
  1405→}

  1407→// ─── Inline formatting ───
  1408→
  1409→function formatInline(text: string): React.ReactNode[] {
  1410→  const parts: React.ReactNode[] = [];
  1411→  const regex = /(\*\*(.+?)\*\*|`([^`]+)`)/g;
  1412→  let lastIndex = 0;
  1413→  let match: RegExpExecArray | null;
  1414→  let key = 0;
  1415→
  1416→  while ((match = regex.exec(text)) !== null) {
  1417→    if (match.index > lastIndex)
  1418→      parts.push(<span key={key++}>{text.slice(lastIndex, match.index)}</span>);
  1419→    if (match[2])
  1420→      parts.push(
  1421→        <strong key={key++} className="font-bold text-foreground">
  1422→          {match[2]}
  1423→        </strong>,
  1424→      );
  1425→    else if (match[3])
  1426→      parts.push(
  1427→        <code
  1428→          key={key++}
  1429→          className="px-1.5 py-0.5 rounded-md bg-primary/10 text-primary font-mono text-xs font-bold"
  1430→        >
  1431→          {match[3]}
  1432→        </code>,
  1433→      );
  1434→    lastIndex = match.index + match[0].length;
  1435→  }
  1436→
  1437→  if (lastIndex < text.length) parts.push(<span key={key++}>{text.slice(lastIndex)}</span>);
  1438→  return parts;
  1439→}
