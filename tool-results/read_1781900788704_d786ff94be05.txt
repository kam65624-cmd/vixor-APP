    38→    38→/** Shape returned from Supabase `experiments` table via listExperiments. */
    39→    39→interface ExperimentRecord {
    40→    40→  id: string;
    41→    41→  user_id: string;

    57→    57→/** Derived best-score summary from the experiment result. */
    58→    58→interface BestScoreSummary {
    59→    59→  overall: number;
    60→    60→  grade: string;
    61→    61→  totalReturn: number;
    62→    62→  maxDrawdown: number;
    63→    63→  sharpe: number;
    64→    64→}
    65→    65→
    66→    66→/** Extract a BestScoreSummary from the serialized experiment result. */
    67→    67→function extractBestScore(result: Record<string, unknown> | null): BestScoreSummary | null {
    68→    68→  if (!result) return null;
    69→    69→  const best = result.bestStrategy as Record<string, unknown> | undefined;
    70→    70→  if (!best || !best.score) return null;
    71→    71→  const summary = best.summary as Record<string, unknown> | undefined;
    72→    72→  const overall = typeof best.score === "number" ? best.score : 0;
    73→    73→  const totalReturn = (summary?.totalReturn as number) ?? 0;
    74→    74→  const maxDrawdown = (summary?.maxDrawdown as number) ?? 0;
    75→    75→  const sharpe = (summary?.sharpe as number) ?? 0;
    76→    76→  const grade = overall >= 90 ? "A" : overall >= 80 ? "B" : overall >= 70 ? "C" : overall >= 60 ? "D" : "F";
    77→    77→  return { overall, grade, totalReturn, maxDrawdown, sharpe };
    78→    78→}
    79→    79→
    80→    80→/** Extract elapsed time in ms from result. */
    81→    81→function extractElapsed(result: Record<string, unknown> | null): number | null {
    82→    82→  if (!result) return null;
    83→    83→  const ms = result.elapsedMs as number | undefined;
    84→    84→  return typeof ms === "number" ? ms : null;
    85→    85→}
    86→    86→
    87→    87→/** Extract ranked strategies count from result. */
    88→    88→function extractRankedCount(result: Record<string, unknown> | null): number {
    89→    89→  if (!result) return 0;
    90→    90→  const arr = result.rankedStrategies as unknown[] | undefined;
    91→    91→  return Array.isArray(arr) ? arr.length : 0;
    92→    92→}
