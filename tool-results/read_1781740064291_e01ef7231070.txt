     4→-- =============================================================================
     5→-- 1. Users & Authentication
     6→-- =============================================================================
     7→
     8→CREATE TABLE IF NOT EXISTS qd_users (
     9→    id SERIAL PRIMARY KEY,
    10→    username VARCHAR(50) UNIQUE NOT NULL,
    11→    password_hash VARCHAR(255) NOT NULL,
    12→    email VARCHAR(100) UNIQUE,
    13→    nickname VARCHAR(50),
    14→    avatar VARCHAR(255) DEFAULT '/avatar2.jpg',
    15→    status VARCHAR(20) DEFAULT 'active',  -- active/disabled/pending
    16→    role VARCHAR(20) DEFAULT 'user',       -- admin/manager/user/viewer
    17→    credits DECIMAL(20,2) DEFAULT 0,       -- 积分余额
    18→    vip_expires_at TIMESTAMP,              -- VIP过期时间
    19→    vip_plan VARCHAR(20) DEFAULT '',       -- VIP套餐：monthly/yearly/lifetime
    20→    vip_is_lifetime BOOLEAN DEFAULT FALSE, -- 是否永久会员
    21→    vip_monthly_credits_last_grant TIMESTAMP, -- 永久会员上次发放月度积分时间
    22→    email_verified BOOLEAN DEFAULT FALSE,  -- 邮箱是否已验证
    23→    referred_by INTEGER,                   -- 邀请人ID
    24→    notification_settings TEXT DEFAULT '', -- 用户通知配置 JSON (telegram_chat_id, default_channels等)
    25→    chart_templates TEXT DEFAULT '',      -- 用户图表模板 JSON（指标布局/样式）
    26→    timezone VARCHAR(64) DEFAULT '',       -- IANA 时区标识，空表示跟随客户端/浏览器
    27→    token_version INTEGER DEFAULT 1,       -- Token版本号，用于单一客户端登录控制
    28→    last_login_at TIMESTAMP,
    29→    created_at TIMESTAMP DEFAULT NOW(),
    30→    updated_at TIMESTAMP DEFAULT NOW()
    31→);
    32→
    33→CREATE INDEX IF NOT EXISTS idx_users_referred_by ON qd_users(referred_by);

   225→
   226→-- Add strategy_mode and strategy_code columns (script strategy support)
   227→DO $$
   228→BEGIN
   229→    IF NOT EXISTS (
   230→        SELECT 1 FROM information_schema.columns
   231→        WHERE table_name = 'qd_strategies_trading' AND column_name = 'strategy_mode'
   232→    ) THEN
   233→        ALTER TABLE qd_strategies_trading ADD COLUMN strategy_mode VARCHAR(20) DEFAULT 'signal';
   234→        RAISE NOTICE 'Added strategy_mode column to qd_strategies_trading';
   235→    END IF;
   236→    IF NOT EXISTS (
   237→        SELECT 1 FROM information_schema.columns
   238→        WHERE table_name = 'qd_strategies_trading' AND column_name = 'strategy_code'
   239→    ) THEN
   240→        ALTER TABLE qd_strategies_trading ADD COLUMN strategy_code TEXT DEFAULT '';
   241→        RAISE NOTICE 'Added strategy_code column to qd_strategies_trading';
   242→    END IF;
   243→END$$;
   244→
   245→-- Add last_rebalance_at column for cross-sectional strategies (if not exists)
   246→DO $$
   247→BEGIN
   248→    IF NOT EXISTS (
   249→        SELECT 1 FROM information_schema.columns 
   250→        WHERE table_name = 'qd_strategies_trading' AND column_name = 'last_rebalance_at'
   251→    ) THEN
   252→        ALTER TABLE qd_strategies_trading ADD COLUMN last_rebalance_at TIMESTAMP;
   253→        RAISE NOTICE 'Added last_rebalance_at column to qd_strategies_trading';
   254→    END IF;
   255→END $$;
