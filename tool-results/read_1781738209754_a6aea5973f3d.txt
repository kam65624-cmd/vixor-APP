   920→// ═══════════════════════════════════════════════
   921→
   922→function formatTimeAgo(
   923→  timestamp: number,
   924→  t?: (key: string, params?: Record<string, string | number>) => string,
   925→) {
   926→  const diff = Date.now() - timestamp;
   927→  const mins = Math.floor(diff / 60000);
   928→  if (mins < 1) return t ? t("common.justNow") : "Just now";
   929→  if (mins < 60) return t ? t("common.mAgo", { mins }) : `${mins}m ago`;
   930→  const hours = Math.floor(mins / 60);
   931→  if (hours < 24) return t ? t("common.hAgo", { hours }) : `${hours}h ago`;
   932→  const days = Math.floor(hours / 24);
   933→  return t ? t("common.dAgo", { days }) : `${days}d ago`;
   934→}
   935→
