  1171→function formatPnl(value: number): string {
  1172→  const prefix = value >= 0 ? "+" : "";
  1173→  return `${prefix}$${Math.abs(value).toFixed(2)}`;
  1174→}

  1176→function formatDuration(hours: number): string {
  1177→  if (hours < 1) return `${Math.round(hours * 60)}m`;
  1178→  if (hours < 24) return `${Math.round(hours * 10) / 10}h`;
  1179→  const days = Math.round((hours / 24) * 10) / 10;
  1180→  return `${days}d`;
  1181→}

  1183→function getBestDay(days: { day: string; winRate: number }[]): string {
  1184→  if (days.length === 0) return "—";
  1185→  const best = days.reduce((a, b) => (a.winRate > b.winRate ? a : b));
  1186→  return best.day;
  1187→}
