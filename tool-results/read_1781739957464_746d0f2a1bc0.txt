     4→class StrategyCompiler:
     5→    def compile(self, config: Dict[str, Any]) -> str:
     6→        """
     7→        Compiles the strategy configuration JSON into executable Python code.
     8→        """
     9→        
    10→        # Extract configurations
    11→        name = config.get('name', 'Generated Strategy')
    12→        entry_rules = config.get('entry_rules', [])
    13→        position_config = config.get('position_config', {})
    14→        pyramiding_rules = config.get('pyramiding_rules', {})
    15→        risk_management = config.get('risk_management', {})
    16→        
    17→        # 1. Imports and Setup
    18→        code = self._get_header(name)
    19→        
    20→        # 2. Parameters (Variables)
    21→        code += self._get_parameters(position_config, pyramiding_rules, risk_management)
    22→        
    23→        # 3. Indicators Calculation
    24→        code += self._get_indicators_calculation(entry_rules)
    25→        
    26→        # 4. Signal Logic (Entry Conditions)
    27→        code += self._get_entry_logic(entry_rules)
    28→        
    29→        # 5. Core Loop (Position Management) - Based on code2.py
    30→        code += self._get_core_loop(position_config, pyramiding_rules, risk_management)
    31→        
    32→        # 6. Output Formatting
    33→        code += self._get_output_section(name, entry_rules)
    34→        
    35→        return code

    54→        
    55→        pyr_enabled = pyr_rules.get('enabled', False)

    86→    def _get_indicators_calculation(self, rules):

    88→# ===========================

    90→# ===========================
    91→"""
    92→        calculated = set()
    93→        
    94→        for rule in rules:
    95→            ind = rule.get('indicator')
    96→            params = rule.get('params', {})
    97→            
    98→            if ind == 'supertrend':
    99→                key = f"st_{params.get('period')}_{params.get('multiplier')}"
   100→                if key not in calculated:
   101→                    code += f"""
   102→# SuperTrend ({params.get('period')}, {params.get('multiplier')})
   103→period = {params.get('period', 14)}
   104→multiplier = {params.get('multiplier', 3.0)}
   105→df['hl2'] = (df['high'] + df['low']) / 2
   106→df['tr'] = np.maximum(df['high'] - df['low'], np.maximum(abs(df['high'] - df['close'].shift(1)), abs(df['low'] - df['close'].shift(1))))
   107→df['atr'] = df['tr'].ewm(alpha=1/period, adjust=False).mean()
   108→df['basic_upper'] = df['hl2'] + (multiplier * df['atr'])
   109→df['basic_lower'] = df['hl2'] - (multiplier * df['atr'])
   110→
   111→final_upper = [0.0] * len(df)
   112→final_lower = [0.0] * len(df)
   113→trend = [1] * len(df)
   114→close_arr = df['close'].values
   115→basic_upper = np.nan_to_num(df['basic_upper'].values)
   116→basic_lower = np.nan_to_num(df['basic_lower'].values)
   117→
   118→for i in range(1, len(df)):
   119→    if basic_upper[i] < final_upper[i-1] or close_arr[i-1] > final_upper[i-1]:
   120→        final_upper[i] = basic_upper[i]
   121→    else:
   122→        final_upper[i] = final_upper[i-1]
   123→        
   124→    if basic_lower[i] > final_lower[i-1] or close_arr[i-1] < final_lower[i-1]:
   125→        final_lower[i] = basic_lower[i]
   126→    else:
   127→        final_lower[i] = final_lower[i-1]
   128→        
   129→    prev_trend = trend[i-1]
   130→    if prev_trend == -1 and close_arr[i] > final_upper[i-1]:
   131→        trend[i] = 1
   132→    elif prev_trend == 1 and close_arr[i] < final_lower[i-1]:
   133→        trend[i] = -1
   134→    else:
   135→        trend[i] = prev_trend
   136→
   137→df['st_trend'] = trend
   138→df['st_upper'] = final_upper
   139→df['st_lower'] = final_lower
   140→"""
   141→                    calculated.add(key)
   142→
   143→            elif ind == 'ema':
   144→                period = params.get('period', 20)
   145→                key = f"ema_{period}"
   146→                if key not in calculated:
   147→                    code += f"\ndf['ema_{period}'] = df['close'].ewm(span={period}, adjust=False).mean()\n"

   150→            elif ind == 'rsi':

   157→gain = (delta.where(delta > 0, 0)).rolling(window={period}).mean()
   158→loss = (-delta.where(delta < 0, 0)).rolling(window={period}).mean()

   162→                    calculated.add(key)
   163→
   164→            elif ind == 'macd':

   378→    def _get_core_loop(self, pos_config, pyr_rules, risk_mgmt):
   379→        # This mirrors the logic in code2.py loop
   380→        return """
   381→# ===========================
   382→# 4. Core Loop (Backtest)
   383→# ===========================
   384→open_long_signals = [False] * len(df)
   385→add_long_signals = [False] * len(df)
   386→close_long_signals = [False] * len(df)
   387→
   388→open_long_text = [None] * len(df)
   389→add_long_text = [None] * len(df)
   390→
   391→open_long_price = [0.0] * len(df)
   392→add_long_price = [0.0] * len(df)
   393→close_long_price = [0.0] * len(df)
   394→close_long_text = [None] * len(df)
   395→
   396→open_short_signals = [False] * len(df)
   397→add_short_signals = [False] * len(df)
   398→close_short_signals = [False] * len(df)
   399→
   400→open_short_price = [0.0] * len(df)
   401→add_short_price = [0.0] * len(df)
   402→close_short_price = [0.0] * len(df)
   403→close_short_text = [None] * len(df)
   404→
   405→position = 0 # 0, 1 (Long), -1 (Short)
   406→position_count = 0
   407→avg_entry_price = 0.0
   408→last_add_price = 0.0
   409→highest_price = 0.0 # For Long: Highest High; For Short: Lowest Low
   410→
   411→close_arr = df['close'].values
   412→high_arr = df['high'].values
   413→low_arr = df['low'].values
   414→raw_buy_arr = df['raw_buy'].values
   415→raw_sell_arr = df['raw_sell'].values
   416→
   417→for i in range(len(df)):
   418→    current_close = close_arr[i]
   419→    current_high = high_arr[i]
   420→    current_low = low_arr[i]
   421→    
   422→    if position == 1:
   423→        # Long Position
   424→        if current_high > highest_price:
   425→            highest_price = current_high
   426→            
   427→        profit_pct = (highest_price - avg_entry_price) / avg_entry_price
   428→        current_profit_pct = (current_close - avg_entry_price) / avg_entry_price
   429→        
   430→        # 1. Trailing Stop
   431→        if take_profit_activation > 0 and profit_pct >= take_profit_activation:
   432→            drawdown = (highest_price - current_close) / avg_entry_price
   433→            if drawdown >= trailing_callback:
   434→                close_long_signals[i] = True
   435→                close_long_price[i] = current_close
   436→                close_long_text[i] = "Trailing Stop"
   437→                position = 0
   438→                position_count = 0
   439→                continue
   440→                
   441→        # 2. Stop Loss
   442→        if stop_loss_pct > 0:
   443→            loss_pct = (avg_entry_price - current_low) / avg_entry_price
   444→            if loss_pct >= stop_loss_pct:
   445→                close_long_signals[i] = True
   446→                close_long_price[i] = avg_entry_price * (1 - stop_loss_pct)
   447→                close_long_text[i] = "Stop Loss"
   448→                position = 0
   449→                position_count = 0
   450→                continue
   451→                
   452→        # 3. Signal Exit (if enabled)
   453→        # Note: Code2 uses raw_sell_arr for exit
   454→        if raw_sell_arr[i]:
   455→             close_long_signals[i] = True
   456→             close_long_price[i] = current_close
   457→             close_long_text[i] = "Signal Exit"
   458→             position = 0
   459→             position_count = 0
   460→             
   461→             # Reverse to Short if trade_direction allows (simplified here)
   462→             # For now we just close.
   463→             continue
   464→             
   465→        # 4. Pyramiding (Add Long)
   466→        if max_pyramiding > 0 and position_count < max_pyramiding + 1 and current_profit_pct > 0:
   467→             # Condition: Price rise by threshold
   468→             if add_threshold_pct > 0:
   469→                 target_price = last_add_price * (1 + add_threshold_pct)
   470→                 if current_high >= target_price:
   471→                     add_long_signals[i] = True
   472→                     add_long_price[i] = target_price
   473→                     add_long_text[i] = "Add Long"
   474→                     position_count += 1
   475→                     last_add_price = target_price
   476→             
   477→    elif position == -1:
   478→        # Short Position
   479→        # For Short, highest_price tracks the LOWEST price (best profit scenario)
   480→        if highest_price == 0: highest_price = avg_entry_price
   481→        if current_low < highest_price:
   482→            highest_price = current_low
   483→            
   484→        # Profit: (Entry - Lowest) / Entry
   485→        profit_pct = (avg_entry_price - highest_price) / avg_entry_price
   486→        current_profit_pct = (avg_entry_price - current_close) / avg_entry_price
   487→        
   488→        # 1. Trailing Stop
   489→        if take_profit_activation > 0 and profit_pct >= take_profit_activation:
   490→            # Drawdown: (Current - Lowest) / Entry
   491→            drawdown = (current_close - highest_price) / avg_entry_price
   492→            if drawdown >= trailing_callback:
   493→                close_short_signals[i] = True
   494→                close_short_price[i] = current_close
   495→                close_short_text[i] = "Trailing Stop"
   496→                position = 0
   497→                position_count = 0
   498→                continue
   499→
   500→        # 2. Stop Loss
   501→        if stop_loss_pct > 0:
   502→            # Loss: Price went up. (High - Entry) / Entry
   503→            loss_pct = (current_high - avg_entry_price) / avg_entry_price
   504→            if loss_pct >= stop_loss_pct:
   505→                close_short_signals[i] = True
   506→                close_short_price[i] = avg_entry_price * (1 + stop_loss_pct)
   507→                close_short_text[i] = "Stop Loss"
   508→                position = 0
   509→                position_count = 0
   510→                continue
   511→
   512→        # 3. Signal Exit
   513→        if raw_buy_arr[i]:
   514→             close_short_signals[i] = True
   515→             close_short_price[i] = current_close
   516→             close_short_text[i] = "Signal Exit"
   517→             position = 0
   518→             position_count = 0
   519→             continue
   520→
   521→        # 4. Pyramiding (Add Short)
   522→        if max_pyramiding > 0 and position_count < max_pyramiding + 1 and current_profit_pct > 0:
   523→             # Condition: Price drop by threshold
   524→             if add_threshold_pct > 0:
   525→                 target_price = last_add_price * (1 - add_threshold_pct)
   526→                 if current_low <= target_price:
   527→                     add_short_signals[i] = True
   528→                     add_short_price[i] = target_price
   529→                     add_short_text[i] = "Add Short"
   530→                     position_count += 1
   531→                     last_add_price = target_price
   532→
   533→    else:
   534→        # No Position
   535→        if raw_buy_arr[i]:
   536→            open_long_signals[i] = True
   537→            open_long_price[i] = current_close
   538→            open_long_text[i] = "Open Long"
   539→            position = 1
   540→            position_count = 1
   541→            avg_entry_price = current_close
   542→            last_add_price = current_close
   543→            highest_price = current_close
   544→            
   545→        elif raw_sell_arr[i]:
   546→            open_short_signals[i] = True
   547→            open_short_price[i] = current_close
   548→            open_short_text[i] = "Open Short"
   549→            position = -1
   550→            position_count = 1
   551→            avg_entry_price = current_close
   552→            last_add_price = current_close
   553→            highest_price = current_close # Init with Entry
   554→
   555→# Append columns
   556→df['open_long'] = open_long_signals
   557→df['add_long'] = add_long_signals
   558→df['close_long'] = close_long_signals
   559→df['open_long_price'] = [p if s else None for p, s in zip(open_long_price, open_long_signals)]
   560→df['add_long_price'] = [p if s else None for p, s in zip(add_long_price, add_long_signals)]
   561→df['close_long_price'] = [p if s else None for p, s in zip(close_long_price, close_long_signals)]
   562→df['open_long_text'] = open_long_text
   563→df['add_long_text'] = add_long_text
   564→df['close_long_text'] = close_long_text
   565→"""
